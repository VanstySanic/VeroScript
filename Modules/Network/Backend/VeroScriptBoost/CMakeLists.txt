cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	project(VeroScriptBoost VERSION 0.1.1 LANGUAGES C CXX)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
	include(CTest)
endif()
if(EMSCRIPTEN)
	message(FATAL_ERROR "The Boost native backend is unavailable in a browser")
endif()
if(NOT TARGET VeroScriptNetwork)
	set(VEROSCRIPT_BUILD_BOOST OFF)
	set(VEROSCRIPT_BUILD_CURL OFF)
	set(_backend_build_testing ${BUILD_TESTING})
	set(BUILD_TESTING OFF)
	add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Modules/Network" "${CMAKE_CURRENT_BINARY_DIR}/Contracts")
	set(BUILD_TESTING ${_backend_build_testing})
endif()
find_package(Threads REQUIRED)
include(CMakePackageConfigHelpers)
include(Resources/Tools/CMake/BoostAsio.cmake)
include("${VEROSCRIPT_MODULE_ROOT}/Modules/Network/VeroScriptNetwork/Resources/Tools/CMake/VeroScriptNetworkProvider.cmake")
set(VEROSCRIPT_BOOST_HTTP OFF)
set(VEROSCRIPT_BOOST_WEBSOCKET OFF)
if(TARGET VeroScriptHttp)
	set(VEROSCRIPT_BOOST_HTTP ON)
endif()
if(TARGET VeroScriptWebSocket)
	set(VEROSCRIPT_BOOST_WEBSOCKET ON)
endif()
set(VEROSCRIPT_BOOST_HAS_HTTP_SERVER ${VEROSCRIPT_BOOST_HTTP} CACHE INTERNAL "Actual Boost HTTP server service" FORCE)
set(VEROSCRIPT_BOOST_HAS_WEBSOCKET ${VEROSCRIPT_BOOST_WEBSOCKET} CACHE INTERNAL "Actual Boost WebSocket services" FORCE)
if(VEROSCRIPT_NETWORK_TLS)
	if(WIN32 AND NOT DEFINED OPENSSL_USE_STATIC_LIBS)
		set(OPENSSL_USE_STATIC_LIBS ON)
	endif()
	find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
endif()
set(_runtime_kind SHARED)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
	set(_runtime_kind STATIC)
endif()
add_library(VeroScriptBoostRuntime ${_runtime_kind} Private/Runtime/NetworkAsio.cpp
	Private/Runtime/NetworkNativeStream.cpp Private/Runtime/BoostRuntime.cpp)
add_library(VeroScript::BoostRuntime ALIAS VeroScriptBoostRuntime)
target_include_directories(VeroScriptBoostRuntime PRIVATE Internal "${VEROSCRIPT_MODULE_ROOT}/Modules/Network/VeroScriptNetwork/Public")
target_include_directories(VeroScriptBoostRuntime SYSTEM PRIVATE ThirdParty)
target_compile_features(VeroScriptBoostRuntime PRIVATE cxx_std_17)
target_compile_definitions(VeroScriptBoostRuntime PRIVATE VEROSCRIPT_NETWORK_RUNTIME_BUILD
	VEROSCRIPT_NETWORK_TLS=$<BOOL:${VEROSCRIPT_NETWORK_TLS}>)
target_link_libraries(VeroScriptBoostRuntime PRIVATE Threads::Threads)
if(VEROSCRIPT_NETWORK_TLS)
	target_link_libraries(VeroScriptBoostRuntime PRIVATE OpenSSL::SSL OpenSSL::Crypto)
	if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
		target_link_libraries(VeroScriptBoostRuntime PRIVATE
			"$<LINK_LIBRARY:FRAMEWORK,Security>" "$<LINK_LIBRARY:FRAMEWORK,CoreFoundation>")
	endif()
endif()
if(WIN32)
	target_link_libraries(VeroScriptBoostRuntime PRIVATE ws2_32 mswsock)
	if(VEROSCRIPT_NETWORK_TLS)
		target_link_libraries(VeroScriptBoostRuntime PRIVATE crypt32)
	endif()
endif()
veroscript_network_asio(VeroScriptBoostRuntime)
set_target_properties(VeroScriptBoostRuntime PROPERTIES EXPORT_NAME BoostRuntime POSITION_INDEPENDENT_CODE ON
	VERSION 0.1.1 SOVERSION 0.1.1 CXX_VISIBILITY_PRESET default)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/veroscript-boost-runtime-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:VeroScriptBoostRuntime>\n")
add_library(VeroScriptBoost STATIC Private/VeroScriptBoost.cpp Private/NetworkTcp.cpp Private/NetworkUdp.cpp Private/BoostUpgrade.cpp)
add_library(VeroScript::Boost ALIAS VeroScriptBoost)
target_include_directories(VeroScriptBoost PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>"
	"$<INSTALL_INTERFACE:include/VeroScriptBoost>" PRIVATE Private Internal)
target_include_directories(VeroScriptBoost SYSTEM PRIVATE ThirdParty)
target_link_libraries(VeroScriptBoost PUBLIC VeroScriptNetwork VeroScriptBoostRuntime PRIVATE VeroScriptNetworkProvider)
target_compile_features(VeroScriptBoost PUBLIC cxx_std_17)
target_compile_definitions(VeroScriptBoost PRIVATE VEROSCRIPT_BOOST_HTTP=$<BOOL:${VEROSCRIPT_BOOST_HTTP}>
	VEROSCRIPT_BOOST_WEBSOCKET=$<BOOL:${VEROSCRIPT_BOOST_WEBSOCKET}>
	VEROSCRIPT_WEBSOCKET_COMPRESSION=$<BOOL:${VEROSCRIPT_WEBSOCKET_COMPRESSION}>)
if(VEROSCRIPT_BOOST_HTTP)
	target_sources(VeroScriptBoost PRIVATE Private/HttpServer.cpp)
	target_link_libraries(VeroScriptBoost PUBLIC VeroScriptHttp)
endif()
if(VEROSCRIPT_BOOST_WEBSOCKET)
	target_sources(VeroScriptBoost PRIVATE Private/BoostWebSocket.cpp)
	target_include_directories(VeroScriptBoost PRIVATE "${VEROSCRIPT_MODULE_ROOT}/Modules/Network/VeroScriptWebSocket/Internal")
	target_link_libraries(VeroScriptBoost PUBLIC VeroScriptWebSocket)
endif()
veroscript_network_asio(VeroScriptBoost)
set_target_properties(VeroScriptBoost PROPERTIES EXPORT_NAME Boost POSITION_INDEPENDENT_CODE ON)
if(WIN32)
	target_link_libraries(VeroScriptBoost PRIVATE ws2_32 mswsock)
endif()
veroscript_network_provider(VeroScriptBoost)
install(TARGETS VeroScriptBoost VeroScriptBoostRuntime EXPORT VeroScriptBoostTargets
	ARCHIVE DESTINATION lib COMPONENT VeroScriptBoost
	LIBRARY DESTINATION lib/veroscript/modules COMPONENT VeroScriptBoost
	RUNTIME DESTINATION lib/veroscript/modules COMPONENT VeroScriptBoost)
vs_install_windows_runtime(lib/veroscript/modules VeroScriptBoost)
install(FILES Public/VeroScriptBoost.h DESTINATION include/VeroScriptBoost COMPONENT VeroScriptBoost)
install(EXPORT VeroScriptBoostTargets NAMESPACE VeroScript:: DESTINATION lib/cmake/VeroScriptBoost COMPONENT VeroScriptBoost)
install(FILES ThirdParty/boost/LICENSE_1_0.txt DESTINATION licenses/VeroScriptBoost/boost COMPONENT VeroScriptBoost)
install(FILES ThirdParty/sources.json DESTINATION licenses/VeroScriptBoost COMPONENT VeroScriptBoost)
install(FILES ThirdParty/patches/boost-websocket-compression.patch DESTINATION licenses/VeroScriptBoost/patches COMPONENT VeroScriptBoost)
if(VEROSCRIPT_NETWORK_TLS)
	install(FILES Resources/Licenses/OpenSSL-LICENSE.txt ThirdParty/openssl-source.json
		DESTINATION licenses/VeroScriptBoost/openssl COMPONENT VeroScriptBoost)
endif()
configure_package_config_file(Resources/Tools/CMake/VeroScriptBoostConfig.cmake.in
	"${CMAKE_CURRENT_BINARY_DIR}/VeroScriptBoostConfig.cmake" INSTALL_DESTINATION lib/cmake/VeroScriptBoost)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/VeroScriptBoostConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptBoostConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptBoostConfigVersion.cmake"
	DESTINATION lib/cmake/VeroScriptBoost COMPONENT VeroScriptBoost)
install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION licenses/VeroScriptBoost COMPONENT VeroScriptBoost)
install(DIRECTORY Resources/Docs DESTINATION share/VeroScriptBoost/Resources COMPONENT VeroScriptBoost OPTIONAL)
