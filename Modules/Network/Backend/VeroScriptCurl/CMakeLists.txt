cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	project(VeroScriptCurl VERSION 0.1.1 LANGUAGES C CXX)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
	include(CTest)
endif()
if(EMSCRIPTEN)
	message(FATAL_ERROR "The native curl backend is unavailable in a browser")
endif()
if(NOT TARGET VeroScriptHttp)
	set(VEROSCRIPT_BUILD_BOOST OFF)
	set(VEROSCRIPT_BUILD_CURL OFF)
	set(VEROSCRIPT_BUILD_HTTP ON)
	set(_backend_build_testing ${BUILD_TESTING})
	set(BUILD_TESTING OFF)
	add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Modules/Network" "${CMAKE_CURRENT_BINARY_DIR}/Contracts")
	set(BUILD_TESTING ${_backend_build_testing})
endif()
include(CMakePackageConfigHelpers)
include(Resources/Tools/CMake/VeroScriptCurlDependencies.cmake)
include("${VEROSCRIPT_MODULE_ROOT}/Modules/Network/VeroScriptNetwork/Resources/Tools/CMake/VeroScriptNetworkProvider.cmake")
veroscript_curl_prepare()
get_target_property(_curl_imported CURL::libcurl IMPORTED)
set(VEROSCRIPT_CURL_BUNDLED ON CACHE INTERNAL "curl is built in this dependency graph" FORCE)
if(_curl_imported)
	set(VEROSCRIPT_CURL_BUNDLED OFF CACHE INTERNAL "curl is built in this dependency graph" FORCE)
endif()
add_library(VeroScriptCurl STATIC Private/VeroScriptCurl.cpp Private/HttpClient.cpp)
add_library(VeroScript::Curl ALIAS VeroScriptCurl)
target_compile_features(VeroScriptCurl PUBLIC cxx_std_17)
target_compile_definitions(VeroScriptCurl PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(VeroScriptCurl PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>"
	"$<INSTALL_INTERFACE:include/VeroScriptCurl>" PRIVATE Private "${VEROSCRIPT_MODULE_ROOT}/Modules/Network/VeroScriptHttp/Internal")
target_link_libraries(VeroScriptCurl PUBLIC VeroScriptHttp PRIVATE VeroScriptNetworkProvider CURL::libcurl)
set_target_properties(VeroScriptCurl PROPERTIES EXPORT_NAME Curl POSITION_INDEPENDENT_CODE ON)
vs_apply_cxx_defaults(VeroScriptCurl)
veroscript_network_provider(VeroScriptCurl)
install(TARGETS VeroScriptCurl EXPORT VeroScriptCurlTargets ARCHIVE DESTINATION lib COMPONENT VeroScriptCurl)
install(FILES Public/VeroScriptCurl.h DESTINATION include/VeroScriptCurl COMPONENT VeroScriptCurl)
install(EXPORT VeroScriptCurlTargets NAMESPACE VeroScript:: DESTINATION lib/cmake/VeroScriptCurl COMPONENT VeroScriptCurl)
install(FILES ThirdParty/curl/COPYING ThirdParty/curl-source.json DESTINATION licenses/VeroScriptCurl/curl COMPONENT VeroScriptCurl)
install(FILES ThirdParty/patches/curl-docs-opt-in.patch DESTINATION licenses/VeroScriptCurl/patches COMPONENT VeroScriptCurl)
configure_package_config_file(Resources/Tools/CMake/VeroScriptCurlConfig.cmake.in
	"${CMAKE_CURRENT_BINARY_DIR}/VeroScriptCurlConfig.cmake" INSTALL_DESTINATION lib/cmake/VeroScriptCurl)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/VeroScriptCurlConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptCurlConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptCurlConfigVersion.cmake"
	DESTINATION lib/cmake/VeroScriptCurl COMPONENT VeroScriptCurl)
install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION licenses/VeroScriptCurl COMPONENT VeroScriptCurl)
install(DIRECTORY Resources/Docs DESTINATION share/VeroScriptCurl/Resources COMPONENT VeroScriptCurl OPTIONAL)
