cmake_minimum_required(VERSION 3.25)
project(VeroScriptNetworkSdkConsumer LANGUAGES CXX)
find_package(VeroScriptNetwork 0.1.1 EXACT REQUIRED)
add_executable(network-sdk-consumer main.cpp)
target_link_libraries(network-sdk-consumer PRIVATE VeroScript::Network)
function(copy_network_runtime target)
	if(WIN32 AND TARGET VeroScript::BoostRuntime)
		get_target_property(_runtime_kind VeroScript::BoostRuntime TYPE)
		if(_runtime_kind STREQUAL "SHARED_LIBRARY")
			add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
				"$<TARGET_FILE:VeroScript::BoostRuntime>" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
		endif()
	endif()
endfunction()
find_package(VeroScriptBoost 0.1.1 EXACT CONFIG QUIET)
find_package(VeroScriptCurl 0.1.1 EXACT CONFIG QUIET)
if(TARGET VeroScript::Boost)
	target_link_libraries(network-sdk-consumer PRIVATE VeroScript::Boost)
	target_compile_definitions(network-sdk-consumer PRIVATE TEST_BOOST=1)
	if(VeroScriptBoost_WEBSOCKET_FOUND)
		target_compile_definitions(network-sdk-consumer PRIVATE TEST_WEBSOCKET=1)
	endif()
	if(TARGET VeroScript::Curl AND VeroScriptBoost_HTTP_SERVER_FOUND)
		target_link_libraries(network-sdk-consumer PRIVATE VeroScript::Curl)
		target_compile_definitions(network-sdk-consumer PRIVATE TEST_HTTP=1)
	endif()
endif()
copy_network_runtime(network-sdk-consumer)
if(MSVC)
	target_compile_options(network-sdk-consumer PRIVATE /EHsc /utf-8)
endif()
enable_testing()
option(VEROSCRIPT_TEST_BINDINGS "Also consume the installed native registration SDK with a matching Core SDK" OFF)
if(VEROSCRIPT_TEST_BINDINGS)
	foreach(component Network Http WebSocket Boost Curl)
		find_package(VeroScript${component} 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Bindings)
	endforeach()
	add_executable(NetworkBindingsConsumer ../NetworkBindingTests.cpp)
	target_link_libraries(NetworkBindingsConsumer PRIVATE VeroScript::NetworkBindings VeroScript::HttpBindings VeroScript::WebSocketBindings VeroScript::BoostBindings VeroScript::CurlBindings)
	target_compile_definitions(NetworkBindingsConsumer PRIVATE VEROSCRIPT_NETWORK_TEST_CURL_PROVIDER=1)
	copy_network_runtime(NetworkBindingsConsumer)
	if(MSVC)
		target_compile_options(NetworkBindingsConsumer PRIVATE /utf-8 /EHsc)
	endif()
	if(MINGW)
		target_link_options(NetworkBindingsConsumer PRIVATE -municode)
	endif()
	add_test(NAME NetworkBindingsConsumer COMMAND NetworkBindingsConsumer)
	set(_modules)
	foreach(_name Network Http WebSocket)
		list(APPEND _modules "${VeroScript${_name}_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScript${_name}Module${CMAKE_SHARED_LIBRARY_SUFFIX}")
	endforeach()
	target_compile_definitions(NetworkBindingsConsumer PRIVATE
		"VEROSCRIPT_TEST_BOOST_MODULE=\"${VeroScriptBoost_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScriptBoostModule${CMAKE_SHARED_LIBRARY_SUFFIX}\""
		"VEROSCRIPT_TEST_CURL_MODULE=\"${VeroScriptCurl_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScriptCurlModule${CMAKE_SHARED_LIBRARY_SUFFIX}\"")
	if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
		set(_backend_modules)
		foreach(_name IN ITEMS Boost Curl)
			list(APPEND _backend_modules "${VeroScript${_name}_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScript${_name}Module${CMAKE_SHARED_LIBRARY_SUFFIX}")
		endforeach()
		foreach(_module IN LISTS _modules _backend_modules)
			if(NOT EXISTS "${_module}")
				message(FATAL_ERROR "Installed dynamic module is missing: ${_module}")
			endif()
		endforeach()
		add_test(NAME NetworkDynamicBindingsConsumer COMMAND NetworkBindingsConsumer ${_modules})
		set_tests_properties(NetworkDynamicBindingsConsumer PROPERTIES TIMEOUT 90)
	endif()
endif()
add_test(NAME NetworkSdkConsumer COMMAND network-sdk-consumer)
