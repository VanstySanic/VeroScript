get_filename_component(_group "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(BUILD_TESTING AND VEROSCRIPT_BUILD_HTTP AND VEROSCRIPT_BUILD_WEBSOCKET AND TARGET VeroScriptBoost AND TARGET VeroScriptCurl)
	find_package(Python3 COMPONENTS Interpreter REQUIRED)
	add_executable(VeroScriptNetworkTests NetworkTests.cpp)
	target_link_libraries(VeroScriptNetworkTests PRIVATE VeroScriptNetwork VeroScriptHttp VeroScriptWebSocket)
	if(MSVC)
		target_compile_options(VeroScriptNetworkTests PRIVATE /EHsc /utf-8)
	endif()
	add_test(NAME VeroScriptNetworkTests COMMAND "${Python3_EXECUTABLE}"
		"${_group}/VeroScriptHttp/Resources/Tests/loopback_runner.py" "$<TARGET_FILE:VeroScriptNetworkTests>")
	set_tests_properties(VeroScriptNetworkTests PROPERTIES TIMEOUT 90 LABELS "VeroScript;Network")
	add_executable(VeroScriptNetworkBackendTimerTests BackendTimerTests.cpp)
	target_link_libraries(VeroScriptNetworkBackendTimerTests PRIVATE VeroScriptNetwork VeroScriptHttp VeroScriptWebSocket)
	if(MSVC)
		target_compile_options(VeroScriptNetworkBackendTimerTests PRIVATE /EHsc /utf-8)
	endif()
	foreach(_mode reconnect download retry)
		add_test(NAME VeroScriptNetworkBackendTimer_${_mode}Tests
			COMMAND VeroScriptNetworkBackendTimerTests "${_mode}")
		set_tests_properties(VeroScriptNetworkBackendTimer_${_mode}Tests PROPERTIES
			TIMEOUT 30 LABELS "VeroScript;Network;Contract;Timer")
	endforeach()
	if(TARGET VeroScriptBoostBindings AND TARGET VeroScriptNetworkBindings AND TARGET VeroScriptHttpBindings AND TARGET VeroScriptWebSocketBindings)
		add_executable(VeroScriptNetworkBindingTests NetworkBindingTests.cpp)
		target_link_libraries(VeroScriptNetworkBindingTests PRIVATE VeroScriptNetworkBindings VeroScriptHttpBindings VeroScriptWebSocketBindings VeroScript)
		if(MSVC)
			target_compile_options(VeroScriptNetworkBindingTests PRIVATE /EHsc /utf-8)
		endif()
		if(MINGW)
			target_link_options(VeroScriptNetworkBindingTests PRIVATE -municode)
		endif()
		if(TARGET VeroScriptNetworkModule AND TARGET VeroScriptHttpModule AND TARGET VeroScriptWebSocketModule)
			add_dependencies(VeroScriptNetworkBindingTests VeroScriptNetworkModule VeroScriptHttpModule VeroScriptWebSocketModule)
			set(_network_test_modules "$<TARGET_FILE:VeroScriptNetworkModule>" "$<TARGET_FILE:VeroScriptHttpModule>" "$<TARGET_FILE:VeroScriptWebSocketModule>")
		else()
			set(_network_test_modules)
		endif()
		if(TARGET VeroScriptAsmJitBackend)
			add_dependencies(VeroScriptNetworkBindingTests VeroScriptAsmJitBackend)
			set(_network_test_jit "$<TARGET_FILE:VeroScriptAsmJitBackend>")
		else()
			set(_network_test_jit "${VEROSCRIPT_NETWORK_TEST_JIT_BACKEND}")
		endif()
		add_test(NAME VeroScriptNetworkBindingTests COMMAND VeroScriptNetworkBindingTests
			${_network_test_modules} ${_network_test_jit})
		set(_network_test_timeout 90)
		if(_network_test_jit)
			set(_network_test_timeout 360)
		endif()
		set_tests_properties(VeroScriptNetworkBindingTests PROPERTIES TIMEOUT ${_network_test_timeout} LABELS "VeroScript;Network;Binding")
	endif()
endif()

if(TARGET VeroScriptBoostModule AND TARGET VeroScriptCurlModule AND TARGET VeroScript
   AND TARGET VeroScriptHttp AND TARGET VeroScriptWebSocket)
	add_executable(VeroScriptNetworkBackendRetirementTests BackendRetirementTests.cpp)
	# Deliberately link only frontends and Core. Providers must really unload.
	target_link_libraries(VeroScriptNetworkBackendRetirementTests PRIVATE
		VeroScriptNetwork VeroScriptHttp VeroScriptWebSocket VeroScript)
	add_dependencies(VeroScriptNetworkBackendRetirementTests VeroScriptBoostModule VeroScriptCurlModule)
	if(MSVC)
		target_compile_options(VeroScriptNetworkBackendRetirementTests PRIVATE /EHsc /utf-8)
	endif()
	if(MINGW)
		target_link_options(VeroScriptNetworkBackendRetirementTests PRIVATE -municode)
	endif()
	add_test(NAME VeroScriptNetworkBackendRetirementTests COMMAND VeroScriptNetworkBackendRetirementTests
		"$<TARGET_FILE:VeroScriptBoostModule>" "$<TARGET_FILE:VeroScriptCurlModule>")
	set_tests_properties(VeroScriptNetworkBackendRetirementTests PROPERTIES
		TIMEOUT 90 LABELS "VeroScript;Network;Lifecycle;Provider")
endif()

get_property(_network_test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(_test IN LISTS _network_test_targets)
	if(_test STREQUAL "VeroScriptNetworkBackendRetirementTests")
		continue()
	endif()
	veroscript_network_test(${_test})
endforeach()
