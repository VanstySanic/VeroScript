cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	project(VeroScriptNetwork VERSION 0.1.1 LANGUAGES C CXX)
	include(CTest)
endif()
set(_vs_network_default_bindings OFF)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../Resources/Tools/CMake/VeroScriptHelpers.cmake")
vs_install_windows_runtime(lib/veroscript/modules VeroScriptNetwork)
if(TARGET VeroScript)
	set(_vs_network_default_bindings ON)
endif()
if(NOT DEFINED VEROSCRIPT_MODULE_ROOT)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
option(VEROSCRIPT_BUILD_BOOST "Build the Boost network provider" ON)
option(VEROSCRIPT_NETWORK_BUILD_BINDINGS "Build script bindings using Core or an installed matching Core SDK" ${_vs_network_default_bindings})
if(VEROSCRIPT_NETWORK_BUILD_BINDINGS AND NOT TARGET VeroScript)
	find_package(VeroScript 0.1.1 EXACT CONFIG REQUIRED)
	add_library(VeroScript ALIAS VeroScript::VeroScript)
endif()
option(VEROSCRIPT_BUILD_HTTP "Build VeroScriptHttp" ON)
option(VEROSCRIPT_BUILD_WEBSOCKET "Build VeroScriptWebSocket" ON)
option(VEROSCRIPT_BUILD_CURL "Build the curl HTTP client provider" ${VEROSCRIPT_BUILD_HTTP})
if(VEROSCRIPT_BUILD_CURL)
	set(VEROSCRIPT_BUILD_HTTP ON)
endif()
option(VEROSCRIPT_WEBSOCKET_COMPRESSION "Enable native permessage-deflate" ON)
set(_vs_network_default_tls ON)
if(CMAKE_SYSTEM_NAME MATCHES "^(iOS|Android)$" OR (NOT VEROSCRIPT_BUILD_HTTP AND NOT VEROSCRIPT_BUILD_WEBSOCKET))
	set(_vs_network_default_tls OFF)
endif()
option(VEROSCRIPT_NETWORK_TLS "Build the native HTTPS server and WSS transport (OpenSSL 3)" ${_vs_network_default_tls})
find_package(Threads REQUIRED)
if(EMSCRIPTEN AND (VEROSCRIPT_BUILD_BOOST OR VEROSCRIPT_BUILD_CURL))
	message(FATAL_ERROR "The native network backend cannot be used in a browser. The browser provider is not implemented yet.")
endif()
if(CMAKE_CXX_FLAGS MATCHES "-fno-exceptions|/EH[^ ]*-")
	message(FATAL_ERROR "VeroScriptNetwork requires C++ exception support")
endif()
include(VeroScriptNetwork/Resources/Tools/CMake/VeroScriptNetworkBuild.cmake)
include(CMakePackageConfigHelpers)
# The versioned provider contract is installed; concrete context layouts remain private.
add_library(VeroScriptNetworkProvider INTERFACE)
target_include_directories(VeroScriptNetworkProvider INTERFACE
	"$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/VeroScriptNetwork/Internal>"
	"$<INSTALL_INTERFACE:include/VeroScriptNetwork/Backend>")
set_target_properties(VeroScriptNetworkProvider PROPERTIES EXPORT_NAME NetworkProvider)
add_library(VeroScriptNetworkBindingSupport INTERFACE)
target_include_directories(VeroScriptNetworkBindingSupport INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/VeroScriptNetwork/Internal/Binding")
add_subdirectory(VeroScriptNetwork)
if(VEROSCRIPT_BUILD_HTTP)
	add_subdirectory(VeroScriptHttp)
endif()
if(VEROSCRIPT_BUILD_WEBSOCKET)
	add_subdirectory(VeroScriptWebSocket)
endif()
foreach(VEROSCRIPT_NETWORK_COMPONENT IN ITEMS Network Http WebSocket)
	set(_package VeroScript${VEROSCRIPT_NETWORK_COMPONENT})
	if(NOT TARGET ${_package})
		continue()
	endif()
	configure_package_config_file("${_package}/Resources/Tools/CMake/${_package}Config.cmake.in"
		"${CMAKE_CURRENT_BINARY_DIR}/${_package}Config.cmake" INSTALL_DESTINATION lib/cmake/${_package})
	write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/${_package}ConfigVersion.cmake"
		VERSION 0.1.1 COMPATIBILITY ExactVersion)
	install(EXPORT ${_package}Targets NAMESPACE VeroScript:: DESTINATION lib/cmake/${_package} COMPONENT ${_package})
	install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${_package}Config.cmake"
		"${CMAKE_CURRENT_BINARY_DIR}/${_package}ConfigVersion.cmake" DESTINATION lib/cmake/${_package} COMPONENT ${_package})
	if(TARGET ${_package}Bindings)
		install(EXPORT ${_package}BindingsTargets NAMESPACE VeroScript:: DESTINATION lib/cmake/${_package}
			COMPONENT ${_package}Bindings)
	endif()
endforeach()
if(TARGET VeroScriptJson)
	foreach(_protocol IN ITEMS Http WebSocket)
		if(TARGET VeroScript${_protocol})
			add_library(VeroScript${_protocol}Json INTERFACE)
			add_library(VeroScript::${_protocol}Json ALIAS VeroScript${_protocol}Json)
			target_link_libraries(VeroScript${_protocol}Json INTERFACE VeroScript${_protocol} VeroScriptJson)
		endif()
	endforeach()
endif()
if(VEROSCRIPT_BUILD_BOOST)
	add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/Backend/VeroScriptBoost" "${CMAKE_CURRENT_BINARY_DIR}/Boost")
endif()
if(VEROSCRIPT_BUILD_CURL AND VEROSCRIPT_BUILD_HTTP)
	add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/Backend/VeroScriptCurl" "${CMAKE_CURRENT_BINARY_DIR}/Curl")
endif()
if(BUILD_TESTING)
	if(TARGET VeroScriptBoost)
		add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/Backend/VeroScriptBoost/Resources/Tests" "${CMAKE_CURRENT_BINARY_DIR}/BoostTests")
	endif()
	add_subdirectory(VeroScriptNetwork/Resources/Tests)
	if(VEROSCRIPT_BUILD_HTTP)
		add_subdirectory(VeroScriptHttp/Resources/Tests)
	endif()
	if(VEROSCRIPT_BUILD_WEBSOCKET)
		add_subdirectory(VeroScriptWebSocket/Resources/Tests)
	endif()
	add_subdirectory(VeroScriptNetwork/Resources/Tests/Integration)
endif()

if(VEROSCRIPT_NETWORK_BUILD_BINDINGS AND
		(TARGET VeroScriptNetworkBindingTests OR TARGET VeroScriptNetworkCompletionTests OR TARGET VeroScriptWebSocketProxyTests))
	if(NOT TARGET VeroScriptBytecodeVmBackend AND
			(NOT EXISTS "${VEROSCRIPT_TEST_VM_LIBRARY}" OR IS_DIRECTORY "${VEROSCRIPT_TEST_VM_LIBRARY}"))
		message(FATAL_ERROR "Network binding tests require a matching VEROSCRIPT_TEST_VM_LIBRARY; use BUILD_TESTING=OFF to build bindings without tests")
	endif()
	if(NOT COMMAND vs_configure_vm_test_directory)
		include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptVmTests.cmake")
	else()
		vs_configure_vm_test_directory("${CMAKE_CURRENT_SOURCE_DIR}")
	endif()
endif()
