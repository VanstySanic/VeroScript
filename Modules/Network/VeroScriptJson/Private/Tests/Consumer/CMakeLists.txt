cmake_minimum_required(VERSION 3.25)
project(VeroScriptJsonSdkConsumer LANGUAGES CXX)
find_package(VeroScriptJson 0.1.1 EXACT REQUIRED)
add_executable(json-sdk-consumer main.cpp)
target_link_libraries(json-sdk-consumer PRIVATE VeroScript::Json)
if(MSVC)
	target_compile_options(json-sdk-consumer PRIVATE /EHsc /utf-8)
endif()
enable_testing()
option(VEROSCRIPT_TEST_BINDINGS "Also consume the installed native registration SDK with a matching Core SDK" OFF)
if(VEROSCRIPT_TEST_BINDINGS)
	find_package(VeroScriptJson 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Bindings)
	add_executable(JsonBindingsConsumer ../JsonBindingTests.cpp)
	target_link_libraries(JsonBindingsConsumer PRIVATE VeroScript::JsonBindings)
	if(MSVC)
		target_compile_options(JsonBindingsConsumer PRIVATE /utf-8 /EHsc)
	endif()
	if(MINGW)
		target_link_options(JsonBindingsConsumer PRIVATE -municode)
	endif()
	add_test(NAME JsonBindingsConsumer COMMAND JsonBindingsConsumer)
	if(NOT EMSCRIPTEN AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
		set(_module "${VeroScriptJson_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScriptJsonModule${CMAKE_SHARED_LIBRARY_SUFFIX}")
		if(NOT EXISTS "${_module}")
			message(FATAL_ERROR "Installed dynamic module is missing: ${_module}")
		endif()
		add_test(NAME JsonDynamicBindingsConsumer COMMAND JsonBindingsConsumer "${_module}")
		set_tests_properties(JsonDynamicBindingsConsumer PROPERTIES TIMEOUT 60)
	endif()
endif()
add_test(NAME JsonSdkConsumer COMMAND json-sdk-consumer)
