cmake_minimum_required(VERSION 3.24)
project(VeroScriptGuiInstalledConsumer LANGUAGES CXX)
find_package(VeroScriptGuiFramework 0.1.1 EXACT CONFIG REQUIRED)
add_executable(GuiInstalledConsumer Main.cpp)
target_link_libraries(GuiInstalledConsumer PRIVATE VeroScript::Gui)
target_compile_options(GuiInstalledConsumer PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
target_compile_definitions(GuiInstalledConsumer PRIVATE
	VS_GUI_TEST_EXAMPLES="${VeroScriptGuiFramework_DIR}/../../../share/VeroScriptGui/Resources/Examples/Markup")
if(TARGET VeroScript::GuiScript)
	find_package(VeroScriptBytecodeVm CONFIG REQUIRED)
	target_link_libraries(GuiInstalledConsumer PRIVATE VeroScript::GuiScript VeroScript::VeroScriptBytecodeVm)
	set(VEROSCRIPT_GUI_TEST_STDLIB "${VeroScript_DIR}/../../../assets/stdlib" CACHE PATH "Matching installed Core standard library")
	if(NOT EXISTS "${VEROSCRIPT_GUI_TEST_STDLIB}/attribute.vs")
		message(FATAL_ERROR "Set VEROSCRIPT_GUI_TEST_STDLIB to the matching Core SDK standard library")
	endif()
	target_compile_definitions(GuiInstalledConsumer PRIVATE VS_GUI_TEST_SCRIPT=1 VS_GUI_TEST_STDLIB="${VEROSCRIPT_GUI_TEST_STDLIB}")
endif()
add_library(GuiInstalledShared SHARED SharedLibrary.cpp)
target_link_libraries(GuiInstalledShared PRIVATE VeroScript::Gui)
set_target_properties(GuiInstalledShared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(GuiInstalledSharedHost SharedMain.cpp)
target_link_libraries(GuiInstalledSharedHost PRIVATE GuiInstalledShared)
find_package(VeroScriptStb 0.1.1 EXACT QUIET CONFIG)
find_package(VeroScriptFontBackend 0.1.1 EXACT QUIET CONFIG)
if(TARGET VeroScript::RmlUi AND TARGET VeroScript::Stb AND TARGET VeroScript::FontBackend)
	target_link_libraries(GuiInstalledConsumer PRIVATE VeroScript::RmlUi VeroScript::Stb VeroScript::FontBackend)
	target_compile_definitions(GuiInstalledConsumer PRIVATE VS_GUI_TEST_RMLUI=1)
	target_link_libraries(GuiInstalledShared PRIVATE VeroScript::RmlUi VeroScript::Stb VeroScript::FontBackend)
	target_compile_definitions(GuiInstalledShared PRIVATE VS_GUI_TEST_RMLUI=1)
endif()
if(TARGET VeroScript::Sdl3)
	target_link_libraries(GuiInstalledConsumer PRIVATE VeroScript::Sdl3)
	target_compile_definitions(GuiInstalledConsumer PRIVATE VS_GUI_TEST_SDL3=1)
endif()
enable_testing()
add_test(NAME GuiInstalledConsumer COMMAND GuiInstalledConsumer)
add_test(NAME GuiInstalledSharedHost COMMAND GuiInstalledSharedHost)
if(TARGET VeroScript::RmlUi)
	set(_vs_gui_test_rmlui ON)
else()
	set(_vs_gui_test_rmlui OFF)
endif()
add_test(NAME GuiInstalledDocumentation COMMAND "${CMAKE_COMMAND}"
	"-DGUI_PREFIX=${VeroScriptGuiFramework_DIR}/../../.."
	"-DGUI_WITH_RMLUI=${_vs_gui_test_rmlui}"
	-P "${CMAKE_CURRENT_LIST_DIR}/Documentation.cmake")
set_tests_properties(GuiInstalledConsumer PROPERTIES TIMEOUT 180)
set_tests_properties(GuiInstalledSharedHost PROPERTIES TIMEOUT 60)
