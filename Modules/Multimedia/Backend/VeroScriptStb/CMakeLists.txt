cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(VeroScriptStb VERSION 0.1.1 LANGUAGES C CXX)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
	include(CTest)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
if(NOT TARGET VeroScript::Image)
	add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Modules/Multimedia/VeroScriptImage" "${CMAKE_CURRENT_BINARY_DIR}/Image")
endif()

add_library(VeroScriptStb STATIC Private/VeroScriptStb.cpp)
add_library(VeroScript::Stb ALIAS VeroScriptStb)
target_link_libraries(VeroScriptStb PUBLIC VeroScript::Image PRIVATE Threads::Threads)
target_include_directories(VeroScriptStb PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>" "$<INSTALL_INTERFACE:include/VeroScriptVisual>" PRIVATE ThirdParty/stb)
target_compile_features(VeroScriptStb PUBLIC cxx_std_17)
set_target_properties(VeroScriptStb PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME Stb)
vs_apply_cxx_defaults(VeroScriptStb)
install(TARGETS VeroScriptStb EXPORT VeroScriptStbTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptStb)
install(EXPORT VeroScriptStbTargets NAMESPACE VeroScript:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptStb" COMPONENT VeroScriptStb)
install(DIRECTORY Public/ DESTINATION include/VeroScriptVisual COMPONENT VeroScriptStb FILES_MATCHING PATTERN "*.h")
include(Resources/Tools/CMake/Provider.cmake)
configure_package_config_file(Resources/Tools/CMake/VeroScriptStbConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptStbConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptStb")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/VeroScriptStbConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptStbConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptStbConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptStb" COMPONENT VeroScriptStb)
install(FILES README.md DESTINATION share/VeroScriptStb COMPONENT VeroScriptStb)
install(FILES ThirdParty/stb/LICENSE ThirdParty/stb.version.json DESTINATION licenses/stb COMPONENT VeroScriptStb)
install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION licenses/VeroScript COMPONENT VeroScriptStb)

if(BUILD_TESTING)
	add_executable(VeroScriptImageTests Resources/Tests/ImageTests.cpp)
	target_link_libraries(VeroScriptImageTests PRIVATE VeroScriptStb Threads::Threads)
	vs_apply_cxx_defaults(VeroScriptImageTests)
	add_test(NAME VeroScriptImageTests COMMAND VeroScriptImageTests)
	set_tests_properties(VeroScriptImageTests PROPERTIES TIMEOUT 30)

	add_executable(VeroScriptImageNoTlsTests Resources/Tests/ImageTests.cpp Private/VeroScriptStb.cpp)
	target_link_libraries(VeroScriptImageNoTlsTests PRIVATE VeroScript::Image Threads::Threads)
	target_include_directories(VeroScriptImageNoTlsTests PRIVATE Public ThirdParty/stb)
	target_compile_definitions(VeroScriptImageNoTlsTests PRIVATE STBI_NO_THREAD_LOCALS)
	vs_apply_cxx_defaults(VeroScriptImageNoTlsTests)
	add_test(NAME VeroScriptImageNoTlsTests COMMAND VeroScriptImageNoTlsTests)
	set_tests_properties(VeroScriptImageNoTlsTests PROPERTIES TIMEOUT 30)
endif()

install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION . COMPONENT VeroScriptStb)
