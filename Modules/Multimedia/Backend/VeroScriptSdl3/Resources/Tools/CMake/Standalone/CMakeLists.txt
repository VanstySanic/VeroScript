cmake_minimum_required(VERSION 3.25)
project(VeroScriptSdl3Native VERSION 0.1.1 LANGUAGES C CXX)
get_filename_component(_vs_sdl3 "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
get_filename_component(VEROSCRIPT_MODULE_ROOT "${_vs_sdl3}/../../../.." ABSOLUTE)
include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
option(VEROSCRIPT_BUILD_AUDIO "Build public Audio contracts" OFF)
option(VEROSCRIPT_BUILD_PLATFORM "Build public Platform contracts" ON)
option(VEROSCRIPT_BUILD_GRAPHICS "Build public Graphics contracts" ON)
option(VEROSCRIPT_BUILD_SDL3 "Build the SDL3 providers" ON)
option(VEROSCRIPT_SDL3_GPU "Build graphics providers" ON)
option(VEROSCRIPT_VISUAL_SCRIPT_BINDINGS "Requires a parent Core build" OFF)
option(BUILD_TESTING "Build native foundation tests" OFF)
option(VEROSCRIPT_VISUAL_WEBGPU_BROWSER_TEST "Build the WebGPU browser regression" OFF)
if(VEROSCRIPT_VISUAL_SCRIPT_BINDINGS)
	message(FATAL_ERROR "Build script adapters from the repository root with the matching Core")
endif()
if(BUILD_TESTING)
	enable_testing()
endif()
if(EMSCRIPTEN)
	add_compile_options(-fexceptions)
	add_link_options(-fexceptions)
endif()
add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/Foundation" "${CMAKE_BINARY_DIR}/Foundation")
if(EMSCRIPTEN AND VEROSCRIPT_VISUAL_WEBGPU_BROWSER_TEST)
	add_executable(VeroScriptVisualWebGpuBrowser
		"${_vs_sdl3}/Resources/Tests/WebGpuBrowser.cpp")
	target_link_libraries(VeroScriptVisualWebGpuBrowser PRIVATE VeroScript::Sdl3)
	target_link_options(VeroScriptVisualWebGpuBrowser PRIVATE -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web)
	set_target_properties(VeroScriptVisualWebGpuBrowser PROPERTIES SUFFIX ".html")
endif()

option(VEROSCRIPT_VISUAL_AUDIO_BROWSER_TEST "Build the browser audio smoke test" OFF)
if(EMSCRIPTEN AND VEROSCRIPT_VISUAL_AUDIO_BROWSER_TEST)
	if(NOT VEROSCRIPT_SDL3_AUDIO OR NOT TARGET VeroScriptAudio)
		message(FATAL_ERROR "Browser audio test requires Audio and the SDL3 audio provider")
	endif()
	add_executable(VeroScriptVisualAudioBrowser
		"${_vs_sdl3}/Resources/Tests/AudioBrowser.cpp")
	target_link_libraries(VeroScriptVisualAudioBrowser PRIVATE VeroScript::Sdl3)
	target_link_options(VeroScriptVisualAudioBrowser PRIVATE -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web)
	set_target_properties(VeroScriptVisualAudioBrowser PROPERTIES SUFFIX ".html")
endif()
