cmake_minimum_required(VERSION 3.25)
project(VeroScriptVisualSdkTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(_vs_sdl3 "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
find_package(VeroScriptVisual 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Sdl3)
if(EMSCRIPTEN)
	add_executable(VeroScriptVisualWebGpuBrowser "${_vs_sdl3}/Resources/Tests/WebGpuBrowser.cpp")
	target_link_libraries(VeroScriptVisualWebGpuBrowser PRIVATE VeroScript::Sdl3)
	target_compile_options(VeroScriptVisualWebGpuBrowser PRIVATE -fexceptions)
	target_link_options(VeroScriptVisualWebGpuBrowser PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web)
	set_target_properties(VeroScriptVisualWebGpuBrowser PROPERTIES SUFFIX ".html"
		RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
	if(VeroScriptVisual_SDL3_AUDIO AND TARGET VeroScript::Audio)
		add_executable(VeroScriptVisualAudioBrowser "${_vs_sdl3}/Resources/Tests/AudioBrowser.cpp")
		target_link_libraries(VeroScriptVisualAudioBrowser PRIVATE VeroScript::Sdl3)
		target_compile_options(VeroScriptVisualAudioBrowser PRIVATE -fexceptions)
		target_link_options(VeroScriptVisualAudioBrowser PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web)
		set_target_properties(VeroScriptVisualAudioBrowser PROPERTIES SUFFIX ".html"
			RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
	endif()
else()
	enable_testing()
	if(TARGET VeroScript::Audio)
		add_executable(VeroScriptSdkAudio "${_vs_sdl3}/Resources/Tests/AudioTests.cpp")
		target_link_libraries(VeroScriptSdkAudio PRIVATE VeroScript::Sdl3 VeroScript::Audio SDL3::SDL3 Threads::Threads)
		target_compile_definitions(VeroScriptSdkAudio PRIVATE VS_TEST_AUDIO=$<BOOL:${VeroScriptVisual_SDL3_AUDIO}>)
		add_test(NAME Audio COMMAND VeroScriptSdkAudio)
		set_tests_properties(Audio PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
		if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND VeroScriptVisual_SDL3_AUDIO)
			add_test(NAME AudioDisconnect COMMAND VeroScriptSdkAudio disconnect)
			set_tests_properties(AudioDisconnect PROPERTIES TIMEOUT 10 SKIP_RETURN_CODE 77
				ENVIRONMENT "SDL_AUDIODRIVER=disk;SDL_AUDIO_DISK_OUTPUT_FILE=/dev/full")
		endif()
	endif()
	add_executable(VeroScriptSdkControllers "${_vs_sdl3}/Resources/Tests/ControllerTests.cpp")
	target_link_libraries(VeroScriptSdkControllers PRIVATE VeroScript::Sdl3 SDL3::SDL3 Threads::Threads)
	target_compile_definitions(VeroScriptSdkControllers PRIVATE VS_TEST_CONTROLLERS=$<BOOL:${VeroScriptVisual_SDL3_CONTROLLERS}>)
	add_test(NAME Controllers COMMAND VeroScriptSdkControllers)
	set_tests_properties(Controllers PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy")
	set(VEROSCRIPT_VISUAL_TEST_DRIVERS "default" CACHE STRING "Semicolon-separated API names present in the installed SDK")
	foreach(_vs_case Graphics2D Composition2D GraphicsAsync)
		add_executable(VeroScriptSdk${_vs_case} "${_vs_sdl3}/Resources/Tests/${_vs_case}Tests.cpp")
		target_link_libraries(VeroScriptSdk${_vs_case} PRIVATE VeroScript::Sdl3)
		foreach(_vs_driver IN LISTS VEROSCRIPT_VISUAL_TEST_DRIVERS)
			if(_vs_driver STREQUAL "default")
				add_test(NAME ${_vs_case}_default COMMAND VeroScriptSdk${_vs_case})
			else()
				add_test(NAME ${_vs_case}_${_vs_driver} COMMAND VeroScriptSdk${_vs_case} ${_vs_driver})
			endif()
			set_tests_properties(${_vs_case}_${_vs_driver} PROPERTIES TIMEOUT 60)
		endforeach()
	endforeach()
endif()
