if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/SDL3/CMakeLists.txt")
	message(FATAL_ERROR "Missing pinned SDL3 source at VeroScriptSdl3/ThirdParty/SDL3")
endif()
# A single shared SDL runtime is used by the platform, graphics and audio adapters. Host builds that
# require static libraries may override SDL_SHARED/SDL_STATIC before configuring.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR EMSCRIPTEN)
	set(_vs_sdl_shared OFF)
else()
	set(_vs_sdl_shared ON)
endif()
option(SDL_SHARED "Build the shared SDL runtime" ${_vs_sdl_shared})
if(SDL_SHARED)
	set(_vs_sdl_static OFF)
else()
	set(_vs_sdl_static ON)
endif()
option(SDL_STATIC "Build the static SDL runtime" ${_vs_sdl_static})
set(SDL_VIDEO ON CACHE BOOL "" FORCE)
set(SDL_GPU ${VEROSCRIPT_SDL3_GPU} CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO ${VEROSCRIPT_SDL3_AUDIO} CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK ${VEROSCRIPT_SDL3_CONTROLLERS} CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL ON CACHE BOOL "Export the single SDL runtime with the Visual SDK" FORCE)
set(SDL_RELOCATABLE ON CACHE BOOL "Keep SDL CMake and pkg-config SDK paths relocatable" FORCE)
set(CMAKE_INSTALL_DEFAULT_COMPONENT_NAME VeroScriptSdl3)
if(TARGET SDL3::SDL3)
	message(FATAL_ERROR "An SDL target already exists: select one source/build owner before adding VeroScriptSdl3")
endif()
add_subdirectory(ThirdParty/SDL3 "${CMAKE_CURRENT_BINARY_DIR}/SDL3")

set(_vs_sdl_sources Private/VeroScriptSdl3.cpp Private/VeroScriptSdl3Runtime.cpp
	Private/VeroScriptSdl3Platform.cpp Private/VeroScriptSdl3WindowInteraction.cpp
	Private/VeroScriptSdl3Input.cpp Private/VeroScriptSdl3Controllers.cpp Private/VeroScriptSdl3Audio.cpp)
if(VEROSCRIPT_SDL3_AUDIO)
	list(APPEND _vs_sdl_sources Private/VeroScriptSdl3AudioDevices.cpp)
endif()
set(_vs_native_extension_default OFF)
if(TARGET VeroScript OR TARGET VeroScript::Core)
	set(_vs_native_extension_default ON)
endif()
option(VEROSCRIPT_SDL3_NATIVE_EXTENSION "Build Core native-extension registration" ${_vs_native_extension_default})
if(VEROSCRIPT_SDL3_NATIVE_EXTENSION)
	if(NOT TARGET VeroScript AND NOT TARGET VeroScript::Core)
		message(FATAL_ERROR "Native extension registration requires the matching Core target and generated ABI headers")
	endif()
	list(APPEND _vs_sdl_sources Private/VeroScriptSdl3Module.cpp)
endif()
if(VEROSCRIPT_SDL3_GRAPHICS)
	list(APPEND _vs_sdl_sources Private/VeroScriptSdl3Graphics.cpp Private/Graphics/VeroScriptSdl3GraphicsBackend.cpp
		Private/Graphics/VeroScriptSdl3D3d11.cpp Private/Graphics/VeroScriptSdl3OpenGl.cpp
		Private/Graphics/VeroScriptSdl3WebGpu.cpp)
endif()
if(VEROSCRIPT_SDL3_GPU)
	list(APPEND _vs_sdl_sources Private/VeroScriptSdl3GraphicsResources.cpp Private/VeroScriptSdl3Renderer2D.cpp)
endif()
if(VEROSCRIPT_SDL3_WEBGPU)
	if(EMSCRIPTEN)
		set(_vs_webgpu_port "--use-port=${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/emdawnwebgpu.remoteport.py")
	else()
		set(VEROSCRIPT_WEBGPU_ROOT "" CACHE PATH "Extracted wgpu-native v29.0.1.1 SDK")
		set(_vs_webgpu_include "${VEROSCRIPT_WEBGPU_ROOT}/include")
		if(NOT EXISTS "${_vs_webgpu_include}/webgpu/wgpu.h" OR
		   NOT EXISTS "${VEROSCRIPT_WEBGPU_ROOT}/wgpu-native-meta/wgpu-native-git-tag")
			message(FATAL_ERROR "Set VEROSCRIPT_WEBGPU_ROOT to the pinned wgpu-native v29.0.1.1 SDK")
		endif()
		file(READ "${VEROSCRIPT_WEBGPU_ROOT}/wgpu-native-meta/wgpu-native-git-tag" _vs_webgpu_tag)
		string(STRIP "${_vs_webgpu_tag}" _vs_webgpu_tag)
		if(NOT _vs_webgpu_tag STREQUAL "v29.0.1.1")
			message(FATAL_ERROR "WebGPU requires wgpu-native v29.0.1.1, got ${_vs_webgpu_tag}")
		endif()
		if(WIN32)
			if(MINGW)
				set(_vs_webgpu_library "${VEROSCRIPT_WEBGPU_ROOT}/lib/libwgpu_native.dll.a")
			else()
				set(_vs_webgpu_library "${VEROSCRIPT_WEBGPU_ROOT}/lib/wgpu_native.dll.lib")
			endif()
			set(_vs_webgpu_runtime "${VEROSCRIPT_WEBGPU_ROOT}/lib/wgpu_native.dll")
		elseif(APPLE)
			set(_vs_webgpu_library "${VEROSCRIPT_WEBGPU_ROOT}/lib/libwgpu_native.dylib")
			set(_vs_webgpu_runtime "${_vs_webgpu_library}")
		else()
			set(_vs_webgpu_library "${VEROSCRIPT_WEBGPU_ROOT}/lib/libwgpu_native.so")
			set(_vs_webgpu_runtime "${_vs_webgpu_library}")
		endif()
		if(NOT EXISTS "${_vs_webgpu_library}" OR NOT EXISTS "${_vs_webgpu_runtime}")
			message(FATAL_ERROR "Set VEROSCRIPT_WEBGPU_ROOT to a matching wgpu-native shared SDK; see Resources/Tools/fetch-webgpu.py")
		endif()
	endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR EMSCRIPTEN)
	add_library(VeroScriptSdl3 STATIC ${_vs_sdl_sources})
	target_compile_definitions(VeroScriptSdl3 PUBLIC VEROSCRIPT_SDL3_STATIC=1)
else()
	add_library(VeroScriptSdl3 SHARED ${_vs_sdl_sources})
endif()
add_library(VeroScript::Sdl3 ALIAS VeroScriptSdl3)
target_include_directories(VeroScriptSdl3 PUBLIC
	"$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>" "$<INSTALL_INTERFACE:include/VeroScriptVisual>")
if(VEROSCRIPT_SDL3_NATIVE_EXTENSION AND TARGET VeroScript)
	target_include_directories(VeroScriptSdl3 PRIVATE "$<TARGET_PROPERTY:VeroScript,INTERFACE_INCLUDE_DIRECTORIES>")
elseif(VEROSCRIPT_SDL3_NATIVE_EXTENSION)
	# Native provider registration uses Core's header-only host contract, never another Core runtime.
	target_include_directories(VeroScriptSdl3 PRIVATE "$<TARGET_PROPERTY:VeroScript::Core,INTERFACE_INCLUDE_DIRECTORIES>")
endif()
target_compile_definitions(VeroScriptSdl3 PUBLIC VEROSCRIPT_SDL3_NATIVE_EXTENSION=$<BOOL:${VEROSCRIPT_SDL3_NATIVE_EXTENSION}>)
target_compile_definitions(VeroScriptSdl3 PRIVATE VEROSCRIPT_SDL3_EXPORTS=1 VEROSCRIPT_SDL3_GPU=$<BOOL:${VEROSCRIPT_SDL3_GPU}>)
target_compile_definitions(VeroScriptSdl3 PRIVATE VEROSCRIPT_SDL3_CONTROLLERS=$<BOOL:${VEROSCRIPT_SDL3_CONTROLLERS}>)
target_compile_definitions(VeroScriptSdl3 PRIVATE VEROSCRIPT_SDL3_GRAPHICS=$<BOOL:${VEROSCRIPT_SDL3_GRAPHICS}>)
target_compile_definitions(VeroScriptSdl3 PRIVATE VEROSCRIPT_SDL3_D3D11=$<BOOL:${VEROSCRIPT_SDL3_D3D11}>
	VEROSCRIPT_SDL3_OPENGL=$<BOOL:${VEROSCRIPT_SDL3_OPENGL}> VEROSCRIPT_SDL3_WEBGPU=$<BOOL:${VEROSCRIPT_SDL3_WEBGPU}>)
target_link_libraries(VeroScriptSdl3 PUBLIC VeroScriptPlatform PRIVATE SDL3::SDL3 Threads::Threads)
target_compile_definitions(VeroScriptSdl3 PRIVATE VEROSCRIPT_SDL3_AUDIO=$<BOOL:${VEROSCRIPT_SDL3_AUDIO}>)
if(VEROSCRIPT_SDL3_AUDIO)
	target_link_libraries(VeroScriptSdl3 PUBLIC VeroScriptAudio)
endif()
if(VEROSCRIPT_SDL3_GRAPHICS)
	target_link_libraries(VeroScriptSdl3 PUBLIC VeroScriptGraphics)
	if(WIN32 AND VEROSCRIPT_SDL3_D3D11)
		target_link_libraries(VeroScriptSdl3 PRIVATE d3d11 dxgi)
	endif()
	if(VEROSCRIPT_SDL3_WEBGPU)
		if(EMSCRIPTEN)
			target_compile_options(VeroScriptSdl3 PRIVATE "${_vs_webgpu_port}")
			target_link_options(VeroScriptSdl3 INTERFACE "$<BUILD_INTERFACE:${_vs_webgpu_port}>"
				"$<INSTALL_INTERFACE:--use-port=$<INSTALL_PREFIX>/share/VeroScriptVisual/emdawnwebgpu.remoteport.py>")
			install(FILES ThirdParty/emdawnwebgpu.remoteport.py DESTINATION share/VeroScriptVisual COMPONENT VeroScriptSdl3)
		else()
			target_include_directories(VeroScriptSdl3 PRIVATE "${_vs_webgpu_include}")
			target_link_libraries(VeroScriptSdl3 PRIVATE "${_vs_webgpu_library}")
			add_custom_command(TARGET VeroScriptSdl3 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
				"${_vs_webgpu_runtime}" "$<TARGET_FILE_DIR:VeroScriptSdl3>")
			if(WIN32)
				install(FILES "${_vs_webgpu_runtime}" DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT VeroScriptSdl3)
			else()
				install(FILES "${_vs_webgpu_runtime}" DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptSdl3)
			endif()
		endif()
	endif()
endif()
set_target_properties(VeroScriptSdl3 PROPERTIES EXPORT_NAME Sdl3 BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN"
	CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
	set_target_properties(VeroScriptSdl3 PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
endif()
if(UNIX AND NOT APPLE)
	target_link_options(VeroScriptSdl3 PRIVATE -Wl,--no-undefined)
	if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
		target_compile_options(VeroScriptSdl3 PRIVATE -fno-gnu-unique)
	endif()
endif()
vs_apply_cxx_defaults(VeroScriptSdl3)
vs_set_solution_folder(VeroScriptSdl3 "Libraries/Backends")
install(TARGETS VeroScriptSdl3 EXPORT VeroScriptVisualSdl3Targets
	RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT VeroScriptSdl3
	LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptSdl3
	ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptSdl3)
install(DIRECTORY Public/ DESTINATION include/VeroScriptVisual COMPONENT VeroScriptSdl3 FILES_MATCHING PATTERN "*.h")
include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptFoundationSdk.cmake")
vs_install_foundation_sdk(Sdl3)
vs_install_windows_runtime("${CMAKE_INSTALL_BINDIR}" VeroScriptSdl3)
install(FILES ThirdParty/SDL3/LICENSE.txt ThirdParty/SDL3.version.json
	DESTINATION licenses/SDL3 COMPONENT VeroScriptSdl3)
install(DIRECTORY Resources/Licenses/ DESTINATION licenses/SDL3 COMPONENT VeroScriptSdl3
	PATTERN "WGPU-NOTICES.txt" EXCLUDE PATTERN "EMDAWN-NOTICES.txt" EXCLUDE)
install(FILES ThirdParty/WgpuNative.version.json DESTINATION licenses/WgpuNative COMPONENT VeroScriptSdl3)
if(VEROSCRIPT_SDL3_WEBGPU)
	if(EMSCRIPTEN)
		install(FILES Resources/Licenses/EMDAWN-NOTICES.txt DESTINATION licenses/Emdawnwebgpu COMPONENT VeroScriptSdl3)
	else()
		install(FILES Resources/Licenses/WGPU-NOTICES.txt DESTINATION licenses/WgpuNative COMPONENT VeroScriptSdl3)
	endif()
endif()
