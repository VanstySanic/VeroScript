# Normal variables are function-scoped. Do not change the host's shared/static library policy.
function(vs_gui_add_rmlui)
	set(BUILD_SHARED_LIBS OFF)
	# Static SDK dependencies must also support linkage into shared libraries.
	set(CMAKE_POSITION_INDEPENDENT_CODE ON)
	set(CMAKE_INSTALL_DEFAULT_COMPONENT_NAME VeroScriptRmlUi)
	set(RMLUI_FONT_ENGINE none)
	set(RMLUI_SAMPLES OFF)
	set(RMLUI_TESTS OFF)
	set(RMLUI_LUA_BINDINGS OFF)
	set(RMLUI_SVG_PLUGIN OFF)
	set(RMLUI_LOTTIE_PLUGIN OFF)
	set(RMLUI_HARFBUZZ_SAMPLE OFF)
	set(RMLUI_TRACY_PROFILING OFF)
	set(RMLUI_INSTALL_TARGETS_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/RmlUi")
	set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
	add_subdirectory(ThirdParty/RmlUi Vendor)
endfunction()
vs_gui_add_rmlui()

		add_library(VeroScriptRmlUiModule SHARED Private/VeroScriptRmlUiModule.cpp)
	endif()

	set_target_properties(VeroScriptRmlUiModule PROPERTIES EXPORT_NAME RmlUiModule POSITION_INDEPENDENT_CODE ON)
	target_include_directories(VeroScriptRmlUiModule PRIVATE "$<TARGET_PROPERTY:${_vs_rml_core},INTERFACE_INCLUDE_DIRECTORIES>")

	target_link_libraries(VeroScriptRmlUiModule PRIVATE VeroScriptRmlUi)
	vs_apply_cxx_defaults(VeroScriptRmlUiModule)
	vs_install_windows_runtime(bin VeroScriptRmlUiBindings)
	vs_set_solution_folder(VeroScriptRmlUiModule "Libraries/Backends")

		RUNTIME DESTINATION bin COMPONENT VeroScriptRmlUiBindings

		ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptRmlUiBindings)
	install(EXPORT VeroScriptRmlUiModuleTargets NAMESPACE VeroScript::
		DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptRmlUi" COMPONENT VeroScriptRmlUiBindings)
endif()
