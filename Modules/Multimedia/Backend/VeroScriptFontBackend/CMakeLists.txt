cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(VeroScriptFontBackend VERSION 0.1.1 LANGUAGES C CXX)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
	include(CTest)
endif()
if(NOT CMAKE_C_COMPILER_LOADED)
	enable_language(C)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET VeroScript::Font)
	add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Modules/Multimedia/VeroScriptFont" "${CMAKE_CURRENT_BINARY_DIR}/Font")
endif()
find_package(Threads REQUIRED)
include(Resources/Tools/CMake/Dependencies.cmake)
add_library(VeroScriptFontBackend STATIC Private/Font/VeroScriptFontBackendFont.cpp
	Private/Layout/VeroScriptFontBackendLayout.cpp)
add_library(VeroScript::FontBackend ALIAS VeroScriptFontBackend)
target_link_libraries(VeroScriptFontBackend PUBLIC VeroScript::Font PRIVATE freetype
	VeroScriptFontBackendHarfBuzz VeroScriptFontBackendSheenBidi Threads::Threads)
target_include_directories(VeroScriptFontBackend PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>"
	"$<INSTALL_INTERFACE:include/VeroScriptVisual>" PRIVATE ThirdParty/HarfBuzz/src ThirdParty/SheenBidi/Headers)
target_compile_features(VeroScriptFontBackend PUBLIC cxx_std_17)
set_target_properties(VeroScriptFontBackend PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME FontBackend)
vs_apply_cxx_defaults(VeroScriptFontBackend)
install(TARGETS VeroScriptFontBackend EXPORT VeroScriptFontBackendTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptFontBackend)
install(EXPORT VeroScriptFontBackendTargets NAMESPACE VeroScript:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFontBackend" COMPONENT VeroScriptFontBackend)
install(DIRECTORY Public/ DESTINATION include/VeroScriptVisual COMPONENT VeroScriptFontBackend FILES_MATCHING PATTERN "*.h")
include(Resources/Tools/CMake/Provider.cmake)
configure_package_config_file(Resources/Tools/CMake/VeroScriptFontBackendConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontBackendConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFontBackend")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontBackendConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontBackendConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontBackendConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFontBackend" COMPONENT VeroScriptFontBackend)
install(FILES README.md DESTINATION share/VeroScriptFontBackend COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/FreeType/docs/FTL.TXT ThirdParty/FreeType/LICENSE.TXT ThirdParty/FreeType.version.json DESTINATION licenses/FreeType COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/FreeType/src/bdf/README DESTINATION licenses/FreeType RENAME LICENSE.BDF.txt COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/FreeType/src/pcf/README DESTINATION licenses/FreeType RENAME LICENSE.PCF.txt COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/FreeType/src/gzip/zlib.h DESTINATION licenses/FreeType RENAME LICENSE.ZLIB.header.txt COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/HarfBuzz/COPYING ThirdParty/HarfBuzz.version.json DESTINATION licenses/HarfBuzz COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/SheenBidi/LICENSE DESTINATION licenses/SheenBidi
	RENAME LICENSE.Apache-2.0.txt COMPONENT VeroScriptFontBackend)
install(FILES ThirdParty/SheenBidi.version.json Resources/Licenses/LICENSE.Unicode-3.0.txt
	Resources/Licenses/NOTICE.SheenBidi.txt DESTINATION licenses/SheenBidi COMPONENT VeroScriptFontBackend)
install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION licenses/VeroScript COMPONENT VeroScriptFontBackend)

if(BUILD_TESTING)
	if(TARGET VeroScript::FontBindings)
		if(TARGET VeroScriptBytecodeVm)
			set(_vs_font_test_vm VeroScriptBytecodeVm)
		else()
			if(NOT TARGET VeroScript::VeroScriptBytecodeVm)
				find_package(VeroScriptBytecodeVm CONFIG REQUIRED)
			endif()
			set(_vs_font_test_vm VeroScript::VeroScriptBytecodeVm)
		endif()
		add_executable(VeroScriptFontBackendScriptBindingsTests Resources/Tests/ScriptBindingsTests.cpp)
		target_link_libraries(VeroScriptFontBackendScriptBindingsTests PRIVATE VeroScriptFontBackend VeroScript::FontBindings ${_vs_font_test_vm})
		vs_apply_cxx_defaults(VeroScriptFontBackendScriptBindingsTests)
		add_test(NAME VeroScriptFontBackendScriptBindingsTests COMMAND VeroScriptFontBackendScriptBindingsTests)
		set_tests_properties(VeroScriptFontBackendScriptBindingsTests PROPERTIES TIMEOUT 30)
	endif()
	add_executable(VeroScriptFontTests Resources/Tests/FontTests.cpp)
	target_link_libraries(VeroScriptFontTests PRIVATE VeroScriptFontBackend)
	vs_apply_cxx_defaults(VeroScriptFontTests)
	add_test(NAME VeroScriptFontTests COMMAND VeroScriptFontTests)
	set_tests_properties(VeroScriptFontTests PROPERTIES TIMEOUT 30)
	add_executable(VeroScriptFontLayoutTests Resources/Tests/LayoutTests.cpp)
	target_link_libraries(VeroScriptFontLayoutTests PRIVATE VeroScriptFontBackend)
	vs_apply_cxx_defaults(VeroScriptFontLayoutTests)
	add_test(NAME VeroScriptFontLayoutTests COMMAND VeroScriptFontLayoutTests
		"${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/SheenBidi/Tools/Unicode/BidiCharacterTest.txt")
	set_tests_properties(VeroScriptFontLayoutTests PROPERTIES TIMEOUT 30)
	if(NOT CMAKE_CROSSCOMPILING)
		set(_vs_dependency_test_args -G "${CMAKE_GENERATOR}")
		if(CMAKE_GENERATOR_PLATFORM)
			list(APPEND _vs_dependency_test_args -A "${CMAKE_GENERATOR_PLATFORM}")
		endif()
		if(CMAKE_GENERATOR_TOOLSET)
			list(APPEND _vs_dependency_test_args -T "${CMAKE_GENERATOR_TOOLSET}")
		endif()
		foreach(_vs_option CMAKE_C_COMPILER CMAKE_CXX_COMPILER CMAKE_MAKE_PROGRAM
				CMAKE_GENERATOR_INSTANCE CMAKE_TOOLCHAIN_FILE)
			if(${_vs_option})
				list(APPEND _vs_dependency_test_args "-D${_vs_option}:FILEPATH=${${_vs_option}}")
			endif()
		endforeach()
		add_test(NAME VeroScriptFontBackendDependencyIsolation COMMAND "${CMAKE_COMMAND}"
			-S "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Tests/DependencyIsolation"
			-B "${CMAKE_CURRENT_BINARY_DIR}/Tests/DependencyIsolation-$<CONFIG>"
			${_vs_dependency_test_args})
		set_tests_properties(VeroScriptFontBackendDependencyIsolation PROPERTIES TIMEOUT 60)
	endif()
endif()

install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION . COMPONENT VeroScriptFontBackend)
