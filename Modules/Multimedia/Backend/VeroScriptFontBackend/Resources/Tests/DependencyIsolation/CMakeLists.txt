cmake_minimum_required(VERSION 3.24)
project(VeroScriptFontBackendDependencyIsolation LANGUAGES C CXX)

get_filename_component(_vs_backend "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
get_filename_component(VEROSCRIPT_MODULE_ROOT "${_vs_backend}/../../../.." ABSOLUTE)
include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")

# Simulate dependencies already discovered by an embedding application.
set(_vs_host_found ZLIB_FOUND BZIP2_FOUND PNG_FOUND HARFBUZZ_FOUND HarfBuzz_FOUND BROTLIDEC_FOUND)
foreach(_vs_found IN LISTS _vs_host_found)
	set(${_vs_found} TRUE)
endforeach()
foreach(_vs_dependency ZLIB BZIP2 PNG BROTLIDEC)
	set(${_vs_dependency}_LIBRARIES "UnexpectedHostDependency_${_vs_dependency}")
endforeach()
set(HarfBuzz_LIBRARY UnexpectedHostDependency_HarfBuzz)
set(BUILD_FRAMEWORK ON)
set(BUILD_SHARED_LIBS ON)

set(BUILD_TESTING OFF)
set(VEROSCRIPT_FONT_BACKEND_NATIVE_EXTENSION OFF)
set(VEROSCRIPT_VISUAL_SCRIPT_BINDINGS OFF)
set(VEROSCRIPT_GUI_SCRIPTING OFF)
add_subdirectory("${_vs_backend}" FontBackend)

get_target_property(_vs_type freetype TYPE)
if(NOT _vs_type STREQUAL "STATIC_LIBRARY")
	message(FATAL_ERROR "The private FreeType dependency must remain static")
endif()
foreach(_vs_property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
	get_target_property(_vs_links freetype ${_vs_property})
	if(_vs_links)
		message(FATAL_ERROR "FreeType inherited host dependencies in ${_vs_property}: ${_vs_links}")
	endif()
endforeach()
file(READ "${CMAKE_CURRENT_BINARY_DIR}/FontBackend/FreeType/include/freetype/config/ftoption.h" _vs_options)
foreach(_vs_option SYSTEM_ZLIB USE_BZIP2 USE_PNG USE_HARFBUZZ USE_BROTLI)
	if(_vs_options MATCHES "(^|\n)[ \t]*#[ \t]*define[ \t]+FT_CONFIG_OPTION_${_vs_option}([ \t\r\n]|$)")
		message(FATAL_ERROR "FreeType inherited the host's FT_CONFIG_OPTION_${_vs_option}")
	endif()
endforeach()
foreach(_vs_found IN LISTS _vs_host_found)
	if(NOT "${${_vs_found}}" STREQUAL "TRUE")
		message(FATAL_ERROR "The dependency changed the host's ${_vs_found}")
	endif()
endforeach()
if(NOT BUILD_FRAMEWORK STREQUAL "ON" OR NOT BUILD_SHARED_LIBS STREQUAL "ON")
	message(FATAL_ERROR "The dependency changed the host's framework/shared-library settings")
endif()
