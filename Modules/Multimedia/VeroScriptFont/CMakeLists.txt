cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(VeroScriptFont VERSION 0.1.1 LANGUAGES CXX)
	get_filename_component(VEROSCRIPT_MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptHelpers.cmake")
	include(CTest)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS Resources/Assets/Fonts/LatoLatin-Regular.ttf)
file(READ Resources/Assets/Fonts/LatoLatin-Regular.ttf VEROSCRIPT_DEFAULT_FONT_BYTES HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," VEROSCRIPT_DEFAULT_FONT_BYTES "${VEROSCRIPT_DEFAULT_FONT_BYTES}")
configure_file(Resources/Tools/CMake/VeroScriptFontDefaultFont.h.in "${CMAKE_CURRENT_BINARY_DIR}/Generated/VeroScriptFontDefaultFont.h" @ONLY)
add_library(VeroScriptFont STATIC Private/VeroScriptFont.cpp Private/VeroScriptFontLayout.cpp
	Private/VeroScriptFontDefaultFont.cpp)
add_library(VeroScript::Font ALIAS VeroScriptFont)
target_link_libraries(VeroScriptFont PRIVATE Threads::Threads)
target_include_directories(VeroScriptFont PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Public>"
	"$<INSTALL_INTERFACE:include/VeroScriptVisual>" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Generated")
target_compile_features(VeroScriptFont PUBLIC cxx_std_17)
set_target_properties(VeroScriptFont PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME Font)
vs_apply_cxx_defaults(VeroScriptFont)
install(TARGETS VeroScriptFont EXPORT VeroScriptFontTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT VeroScriptFont)
install(EXPORT VeroScriptFontTargets NAMESPACE VeroScript:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFont" COMPONENT VeroScriptFont)
install(DIRECTORY Public/ DESTINATION include/VeroScriptVisual COMPONENT VeroScriptFont FILES_MATCHING PATTERN "*.h" PATTERN "VeroScriptFontBindings.h" EXCLUDE)
include(Resources/Tools/CMake/Bindings.cmake)
configure_package_config_file(Resources/Tools/CMake/VeroScriptFontConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFont")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/VeroScriptFontConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/VeroScriptFont" COMPONENT VeroScriptFont)
install(FILES Resources/Assets/Fonts/LICENSE.txt DESTINATION licenses/Fonts RENAME LICENSE.Lato.txt COMPONENT VeroScriptFont)
install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION licenses/VeroScript COMPONENT VeroScriptFont)
install(FILES README.md DESTINATION share/VeroScriptFont COMPONENT VeroScriptFont)

install(FILES "${VEROSCRIPT_MODULE_ROOT}/LICENSE" DESTINATION . COMPONENT VeroScriptFont)

if(BUILD_TESTING)
	add_executable(VeroScriptFontContractTests Resources/Tests/Contracts.cpp)
	target_link_libraries(VeroScriptFontContractTests PRIVATE VeroScriptFont)
	vs_apply_cxx_defaults(VeroScriptFontContractTests)
	add_test(NAME VeroScriptFontContractTests COMMAND VeroScriptFontContractTests)
	set_tests_properties(VeroScriptFontContractTests PROPERTIES TIMEOUT 30)
	add_executable(VeroScriptFontLayoutContractTests Resources/Tests/LayoutContracts.cpp)
	target_link_libraries(VeroScriptFontLayoutContractTests PRIVATE VeroScriptFont)
	vs_apply_cxx_defaults(VeroScriptFontLayoutContractTests)
	add_test(NAME VeroScriptFontLayoutContractTests COMMAND VeroScriptFontLayoutContractTests)
	set_tests_properties(VeroScriptFontLayoutContractTests PROPERTIES TIMEOUT 30)
endif()
