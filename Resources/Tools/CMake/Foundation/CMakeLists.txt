foreach(_vs_resource Image Font Stb FontBackend)
	string(TOUPPER "${_vs_resource}" _vs_resource_option)
	if(_vs_resource STREQUAL "FontBackend")
		set(_vs_resource_option FONT_BACKEND)
	endif()
	if(VEROSCRIPT_BUILD_${_vs_resource_option} AND NOT TARGET VeroScript${_vs_resource})
		if(_vs_resource MATCHES "^(Stb|FontBackend)$")
			set(_vs_resource_group Multimedia/Backend)
		else()
			set(_vs_resource_group Multimedia)
		endif()
		add_subdirectory("${VEROSCRIPT_MODULE_ROOT}/Modules/${_vs_resource_group}/VeroScript${_vs_resource}" ${_vs_resource})
	endif()
endforeach()
foreach(_vs_resource Image Font Stb FontBackend)
	if(TARGET VeroScript${_vs_resource})
		get_target_property(_vs_resource_binary VeroScript${_vs_resource} BINARY_DIR)
		install(CODE "block(SCOPE_FOR VARIABLES PROPAGATE CMAKE_INSTALL_MANIFEST_FILES CMAKE_ABSOLUTE_DESTINATION_FILES)\nset(CMAKE_INSTALL_COMPONENT VeroScript${_vs_resource})\ninclude(\"${_vs_resource_binary}/cmake_install.cmake\")\nendblock()"
			COMPONENT VeroScriptVisual)
		install(CODE "block(SCOPE_FOR VARIABLES PROPAGATE CMAKE_INSTALL_MANIFEST_FILES CMAKE_ABSOLUTE_DESTINATION_FILES)\nset(CMAKE_INSTALL_COMPONENT VeroScript${_vs_resource}Bindings)\ninclude(\"${_vs_resource_binary}/cmake_install.cmake\")\nendblock()"
			COMPONENT VeroScriptVisual)
	endif()
endforeach()
