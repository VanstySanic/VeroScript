cmake_minimum_required(VERSION 3.25)
project(VeroScriptMobileSdkConsumer LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Android|iOS)$")
	message(FATAL_ERROR "Mobile SDK consumers require the actual target toolchain")
endif()
foreach(_argument VERO_SDK_PACKAGE VERO_SDK_TARGET VERO_SDK_HEADER VERO_SDK_VERSION)
	if(NOT DEFINED ${_argument})
		message(FATAL_ERROR "Missing mobile SDK consumer argument: ${_argument}")
	endif()
endforeach()
find_package(${VERO_SDK_PACKAGE} ${VERO_SDK_VERSION} EXACT CONFIG REQUIRED)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" "#include \"${VERO_SDK_HEADER}\"\nint main() { return 0; }\n")
add_executable(VeroScriptMobileSdkConsumer "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
target_compile_features(VeroScriptMobileSdkConsumer PRIVATE cxx_std_17)
set_target_properties(VeroScriptMobileSdkConsumer PROPERTIES MACOSX_BUNDLE OFF)
if(VERO_SDK_VM)
	find_package(VeroScriptBytecodeVm ${VERO_SDK_VERSION} EXACT CONFIG REQUIRED)
endif()
# Retain every object of the selected public library. An empty main linked
# normally would allow a broken archive/dependency graph to pass unnoticed.
foreach(_target ${VERO_SDK_TARGET} ${VERO_SDK_VM})
	get_target_property(_kind ${_target} TYPE)
	if(CMAKE_SYSTEM_NAME STREQUAL "Android")
		if(_kind STREQUAL "STATIC_LIBRARY")
			target_link_libraries(VeroScriptMobileSdkConsumer PRIVATE "-Wl,--whole-archive" ${_target} "-Wl,--no-whole-archive")
		elseif(_kind STREQUAL "SHARED_LIBRARY")
			# Keep the bridge/provider even when the smoke main needs no symbol.
			target_link_libraries(VeroScriptMobileSdkConsumer PRIVATE "-Wl,--no-as-needed" ${_target} "-Wl,--as-needed")
		else()
			message(FATAL_ERROR "Unexpected mobile SDK library kind: ${_kind}")
		endif()
	else()
		if(NOT _kind STREQUAL "STATIC_LIBRARY")
			message(FATAL_ERROR "The iOS SDK requires static libraries")
		endif()
		target_link_options(VeroScriptMobileSdkConsumer PRIVATE "LINKER:-force_load,$<TARGET_FILE:${_target}>")
		target_link_libraries(VeroScriptMobileSdkConsumer PRIVATE ${_target})
	endif()
endforeach()
if(VERO_SDK_OPENSSL)
	find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
	target_link_libraries(VeroScriptMobileSdkConsumer PRIVATE OpenSSL::SSL OpenSSL::Crypto)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Android")
	target_link_options(VeroScriptMobileSdkConsumer PRIVATE "LINKER:--no-undefined" "LINKER:--no-allow-shlib-undefined" "LINKER:-z,max-page-size=16384")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/consumer-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:VeroScriptMobileSdkConsumer>\n")
