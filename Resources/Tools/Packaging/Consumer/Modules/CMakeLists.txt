cmake_minimum_required(VERSION 3.25)
project(VeroScriptModuleConsumer LANGUAGES CXX)
enable_testing()
set(VEROSCRIPT_MODULES "" CACHE STRING "Installed module selection")
option(VEROSCRIPT_TEST_BINDINGS "Consume static registration and dynamic modules" ON)
if(VEROSCRIPT_MODULES STREQUAL "Network")
	set(CMAKE_DISABLE_FIND_PACKAGE_CURL TRUE)
	find_package(VeroScriptNetwork 0.1.1 EXACT CONFIG REQUIRED)
	if(TARGET VeroScript::Http OR TARGET VeroScript::WebSocket)
		message(FATAL_ERROR "Network-only consumption loaded an unrelated protocol")
	endif()
endif()
# Resolve protocols before their foundation to verify transitive package discovery.
set(_find_modules ${VEROSCRIPT_MODULES})
list(REVERSE _find_modules)
foreach(_module IN LISTS _find_modules)
	if(VEROSCRIPT_TEST_BINDINGS)
		find_package(VeroScript${_module} 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Bindings)
	else()
		find_package(VeroScript${_module} 0.1.1 EXACT CONFIG REQUIRED)
	endif()
	get_filename_component(_module_prefix "${VeroScript${_module}_DIR}/../../.." ABSOLUTE)
	get_filename_component(_module_directory "${VeroScript${_module}_MODULE_DIR}" ABSOLUTE)
	if(NOT _module_directory STREQUAL "${_module_prefix}/lib/veroscript/modules")
		message(FATAL_ERROR "VeroScript${_module} resolved its modules directory outside its own SDK: ${_module_directory}")
	endif()
endforeach()
if("Json" IN_LIST VEROSCRIPT_MODULES AND ("Http" IN_LIST VEROSCRIPT_MODULES OR "WebSocket" IN_LIST VEROSCRIPT_MODULES))
	add_executable(json-protocols JsonProtocols.cpp)
	foreach(_module IN ITEMS Http WebSocket)
		if("${_module}" IN_LIST VEROSCRIPT_MODULES)
			find_package(VeroScript${_module} 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Json)
			target_link_libraries(json-protocols PRIVATE VeroScript::${_module}Json)
			string(TOUPPER "${_module}" _define)
			target_compile_definitions(json-protocols PRIVATE TEST_${_define}=1)
		endif()
	endforeach()
	add_test(NAME JsonProtocolConsumer COMMAND json-protocols)
endif()
if("Json" IN_LIST VEROSCRIPT_MODULES)
	add_executable(json-data "${VEROSCRIPT_JSON_CONSUMER_SOURCE}")
	target_link_libraries(json-data PRIVATE VeroScript::Json)
	add_test(NAME JsonDataConsumer COMMAND json-data)
endif()
if("Network" IN_LIST VEROSCRIPT_MODULES)
	add_executable(network-data "${VEROSCRIPT_NETWORK_CONSUMER_SOURCE}")
	target_link_libraries(network-data PRIVATE VeroScript::Network)
	if(TARGET VeroScript::Boost)
		target_link_libraries(network-data PRIVATE VeroScript::Boost)
		target_compile_definitions(network-data PRIVATE TEST_BOOST=1)
	endif()
	if(TARGET VeroScript::Curl)
		target_link_libraries(network-data PRIVATE VeroScript::Curl)
	endif()
	foreach(_module IN ITEMS Http WebSocket)
		if("${_module}" IN_LIST VEROSCRIPT_MODULES AND TARGET VeroScript::Boost AND
		   ((_module STREQUAL "Http" AND TARGET VeroScript::Curl AND VeroScriptBoost_HTTP_SERVER_FOUND) OR
			(_module STREQUAL "WebSocket" AND VeroScriptBoost_WEBSOCKET_FOUND)))
			target_link_libraries(network-data PRIVATE VeroScript::${_module})
			string(TOUPPER "${_module}" _define)
			target_compile_definitions(network-data PRIVATE TEST_${_define}=1)
		endif()
	endforeach()
	add_test(NAME NetworkDataConsumer COMMAND network-data)
endif()
if(VEROSCRIPT_TEST_BINDINGS)
	if(NOT EXISTS "${VEROSCRIPT_TEST_VM_LIBRARY}" OR IS_DIRECTORY "${VEROSCRIPT_TEST_VM_LIBRARY}")
		message(FATAL_ERROR "Script consumers require an explicit matching VEROSCRIPT_TEST_VM_LIBRARY")
	endif()
	add_executable(script-modules ScriptConsumer.cpp)
	if(MINGW)
		target_link_options(script-modules PRIVATE -municode)
	endif()
	set(_dynamic)
	foreach(_module IN LISTS VEROSCRIPT_MODULES)
		target_link_libraries(script-modules PRIVATE VeroScript::${_module}Bindings)
		string(TOUPPER "${_module}" _define)
		target_compile_definitions(script-modules PRIVATE TEST_${_define}=1)
		list(APPEND _dynamic "${VeroScript${_module}_MODULE_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}VeroScript${_module}Module${CMAKE_SHARED_LIBRARY_SUFFIX}")
	endforeach()
	add_test(NAME StaticScriptConsumer COMMAND script-modules "${VEROSCRIPT_TEST_VM_LIBRARY}")
	add_test(NAME DynamicScriptConsumer COMMAND script-modules "${VEROSCRIPT_TEST_VM_LIBRARY}" ${_dynamic})
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../CopyRuntime.cmake")
get_property(_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(_target IN LISTS _targets)
	if(MSVC)
		target_compile_options(${_target} PRIVATE /EHsc /utf-8)
	endif()
	if(WIN32 AND TARGET VeroScript::BoostRuntime)
		veroscript_consumer_copy_runtime(${_target} "$<TARGET_FILE:VeroScript::BoostRuntime>")
	endif()
endforeach()
