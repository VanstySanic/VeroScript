cmake_minimum_required(VERSION 3.25)
project(VeroScriptFoundationConsumer LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VEROSCRIPT_TEST_BINDINGS "Consume the installed script adapters" ON)
if(VEROSCRIPT_TEST_GUI_FRAMEWORK_ONLY)
	find_package(VeroScriptGuiFramework 0.1.1 EXACT CONFIG REQUIRED)
	if(VEROSCRIPT_TEST_BINDINGS AND
			(NOT TARGET VeroScript::GuiScript OR NOT TARGET VeroScript::RmlUiModule))
		message(FATAL_ERROR "The legacy GUI entry did not import the installed script adapters")
	endif()
	if(NOT VEROSCRIPT_TEST_BINDINGS AND
			(TARGET VeroScript::Core OR TARGET VeroScript::GuiScript))
		message(FATAL_ERROR "Native legacy GUI discovery imported script dependencies")
	endif()
	return()
endif()
# Ninja's UTF-8 response files need a BOM when consumed by the MSVC linker.
if(VEROSCRIPT_PACKAGING_PYTHON AND CMAKE_HOST_WIN32 AND MSVC AND
		CMAKE_GENERATOR MATCHES "^Ninja" AND NOT CMAKE_CXX_LINKER_LAUNCHER)
	execute_process(COMMAND "${CMAKE_MAKE_PROGRAM}" -t wincodepage
		RESULT_VARIABLE _vs_ninja_status OUTPUT_VARIABLE _vs_ninja_codepage ERROR_QUIET)
	if(_vs_ninja_status STREQUAL "0" AND _vs_ninja_codepage MATCHES "Build file encoding: UTF-8")
		set(CMAKE_CXX_LINKER_LAUNCHER "${VEROSCRIPT_PACKAGING_PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/LinkMsvc.py")
	endif()
endif()
foreach(_vs_module IN LISTS VEROSCRIPT_FOUNDATION_MODULES)
	if(VEROSCRIPT_TEST_BINDINGS AND NOT _vs_module STREQUAL "Sdl3")
		find_package(VeroScript${_vs_module} 0.1.1 EXACT CONFIG REQUIRED COMPONENTS Bindings)
	else()
		find_package(VeroScript${_vs_module} 0.1.1 EXACT CONFIG REQUIRED)
	endif()
	# Repeated independent discovery must be safe.
	find_package(VeroScript${_vs_module} 0.1.1 EXACT CONFIG REQUIRED)
endforeach()
# Resource fixtures use public SDK factories; retain other modules' existing test setup.
set(_vs_vm_helper_modules ${VEROSCRIPT_FOUNDATION_MODULES})
list(REMOVE_ITEM _vs_vm_helper_modules Image Font Stb FontBackend)
if(VEROSCRIPT_TEST_BINDINGS AND _vs_vm_helper_modules)
	if(NOT EXISTS "${VEROSCRIPT_TEST_VM_LIBRARY}" OR
			NOT EXISTS "${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptVmTests.cmake")
		message(FATAL_ERROR "Script regressions require VEROSCRIPT_MODULE_ROOT and an explicit matching VEROSCRIPT_TEST_VM_LIBRARY")
	endif()
	# These source fixtures use Core's test setup; installed SDKs retain only public dependencies.
	set(BUILD_TESTING ON)
	include("${VEROSCRIPT_MODULE_ROOT}/Resources/Tools/CMake/VeroScriptVmTests.cmake")
endif()
if(NOT VEROSCRIPT_TEST_BINDINGS)
	foreach(_vs_target VeroScript::Core VeroScript::PlatformBindings VeroScript::GraphicsBindings VeroScript::AudioBindings
			VeroScript::ImageBindings VeroScript::FontBindings VeroScript::GuiScript)
		if(TARGET ${_vs_target})
			message(FATAL_ERROR "Native discovery imported unrelated script target ${_vs_target}")
		endif()
	endforeach()
endif()
if("Gui" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
	add_library(gui-shared SHARED GuiShared.cpp)
	set_target_properties(gui-shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
	target_link_libraries(gui-shared PRIVATE VeroScript::Gui)
	if("RmlUi" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
		target_link_libraries(gui-shared PRIVATE VeroScript::RmlUi)
	endif()
	add_executable(gui-shared-host GuiSharedHost.cpp)
	target_link_libraries(gui-shared-host PRIVATE gui-shared)
	add_test(NAME GuiSharedHost COMMAND gui-shared-host)
	if(VEROSCRIPT_TEST_BINDINGS)
		find_package(VeroScriptBytecodeVm CONFIG REQUIRED)
		add_executable(gui-script GuiScript.cpp)
		target_link_libraries(gui-script PRIVATE VeroScript::GuiScript VeroScript::VeroScriptBytecodeVm)
		file(TO_CMAKE_PATH "${VEROSCRIPT_GUI_TEST_STDLIB}" _vs_gui_test_stdlib)
		target_compile_definitions(gui-script PRIVATE VS_GUI_TEST_STDLIB="${_vs_gui_test_stdlib}")
		add_test(NAME GuiScript COMMAND gui-script)
		set_tests_properties(GuiScript PROPERTIES TIMEOUT 300)
	endif()
	if("RmlUi" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
		find_package(VeroScriptGuiFramework 0.1.1 EXACT CONFIG REQUIRED)
	endif()
endif()
if(NOT "Sdl3" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
	foreach(_vs_target VeroScript::Sdl3 SDL3::SDL3)
		if(TARGET ${_vs_target})
			message(FATAL_ERROR "A public service imported its backend ${_vs_target}")
		endif()
	endforeach()
endif()
add_executable(foundation-contracts Contracts.cpp)
foreach(_vs_module IN LISTS VEROSCRIPT_FOUNDATION_MODULES)
	target_link_libraries(foundation-contracts PRIVATE VeroScript::${_vs_module})
	string(TOUPPER "${_vs_module}" _vs_define)
	if(_vs_module STREQUAL "FontBackend")
		set(_vs_define FONT_BACKEND)
	endif()
	target_compile_definitions(foundation-contracts PRIVATE VS_TEST_${_vs_define}=1)
endforeach()
add_test(NAME FoundationContracts COMMAND foundation-contracts)
set_tests_properties(FoundationContracts PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
if(VEROSCRIPT_TEST_BINDINGS)
	if("Image" IN_LIST VEROSCRIPT_FOUNDATION_MODULES OR
			("FontBackend" IN_LIST VEROSCRIPT_FOUNDATION_MODULES AND TARGET VeroScript::FontBindings))
		# Script regressions select their test VM explicitly; the service SDKs do not select one.
		find_package(VeroScriptBytecodeVm CONFIG REQUIRED)
	endif()
	foreach(_vs_provider Stb FontBackend RmlUi)
		if(_vs_provider IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
			set(_vs_provider_source Providers.cpp)
			if(_vs_provider STREQUAL "FontBackend")
				# This module registers two services; reuse its contract/lifetime regression.
				set(_vs_provider_source "${VEROSCRIPT_FOUNDATION_SOURCE_FONTBACKEND}/Resources/Tests/ProviderTests.cpp")
			endif()
			add_executable(provider-${_vs_provider} "${_vs_provider_source}")
			target_link_libraries(provider-${_vs_provider} PRIVATE VeroScript::Core)
			if(_vs_provider STREQUAL "Stb" OR _vs_provider STREQUAL "FontBackend")
				get_target_property(_vs_provider_type VeroScript::${_vs_provider}Module TYPE)
				if(_vs_provider_type STREQUAL "STATIC_LIBRARY")
					target_link_libraries(provider-${_vs_provider} PRIVATE VeroScript::${_vs_provider}Module)
				endif()
			endif()
			if(_vs_provider STREQUAL "FontBackend")
				target_link_libraries(provider-${_vs_provider} PRIVATE VeroScript::Font)
				if(TARGET VeroScript::FontBindings)
					add_executable(font-script "${VEROSCRIPT_FOUNDATION_SOURCE_FONTBACKEND}/Resources/Tests/ScriptBindingsTests.cpp")
					target_link_libraries(font-script PRIVATE VeroScript::FontBackend VeroScript::FontBindings VeroScript::VeroScriptBytecodeVm)
					add_test(NAME ScriptFont COMMAND font-script)
					set_tests_properties(ScriptFont PROPERTIES TIMEOUT 30)
				endif()
			endif()
			if(MINGW)
				target_link_options(provider-${_vs_provider} PRIVATE -municode)
			endif()
			add_test(NAME Provider${_vs_provider} COMMAND provider-${_vs_provider}
				"$<TARGET_FILE:VeroScript::${_vs_provider}Module>")
			set_tests_properties(Provider${_vs_provider} PROPERTIES TIMEOUT 30)
		endif()
	endforeach()
	if("RmlUi" IN_LIST VEROSCRIPT_FOUNDATION_MODULES AND
			"Stb" IN_LIST VEROSCRIPT_FOUNDATION_MODULES AND "FontBackend" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
		add_executable(gui-provider "${VEROSCRIPT_FOUNDATION_SOURCE_RMLUI}/Resources/Tests/ProviderTests.cpp")
		target_link_libraries(gui-provider PRIVATE VeroScript::GuiScript VeroScript::Stb VeroScript::FontBackend)
		if(MINGW)
			target_link_options(gui-provider PRIVATE -municode)
		endif()
		add_test(NAME GuiProvider COMMAND gui-provider "$<TARGET_FILE:VeroScript::RmlUiModule>")
		set_tests_properties(GuiProvider PROPERTIES TIMEOUT 60)
	endif()
	foreach(_vs_module Platform Audio Image)
		if(_vs_module IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
			string(TOUPPER "${_vs_module}" _vs_source_key)
			if(_vs_module STREQUAL "Platform")
				set(_vs_source BackendBindings.cpp)
			elseif(_vs_module STREQUAL "Image")
				set(_vs_source BindingsTests.cpp)
			else()
				set(_vs_source BindingTests.cpp)
			endif()
			add_executable(script-${_vs_module} "${VEROSCRIPT_FOUNDATION_SOURCE_${_vs_source_key}}/Resources/Tests/${_vs_source}")
			target_link_libraries(script-${_vs_module} PRIVATE VeroScript::${_vs_module}Bindings)
			if(_vs_module STREQUAL "Image")
				target_link_libraries(script-${_vs_module} PRIVATE VeroScript::VeroScriptBytecodeVm)
			endif()
			add_test(NAME Script${_vs_module} COMMAND script-${_vs_module})
			set_tests_properties(Script${_vs_module} PROPERTIES TIMEOUT 30)
		endif()
	endforeach()
endif()
if("Sdl3" IN_LIST VEROSCRIPT_FOUNDATION_MODULES)
	add_executable(controllers "${VEROSCRIPT_FOUNDATION_SOURCE_SDL3}/Resources/Tests/ControllerTests.cpp")
	target_link_libraries(controllers PRIVATE VeroScript::Sdl3 SDL3::SDL3 Threads::Threads)
	target_compile_definitions(controllers PRIVATE VS_TEST_CONTROLLERS=$<BOOL:${VeroScriptSdl3_SDL3_CONTROLLERS}>)
	add_test(NAME Controllers COMMAND controllers)
	set_tests_properties(Controllers PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy")
	if(VeroScriptSdl3_SDL3_AUDIO)
		add_executable(audio "${VEROSCRIPT_FOUNDATION_SOURCE_SDL3}/Resources/Tests/AudioTests.cpp")
		target_link_libraries(audio PRIVATE VeroScript::Sdl3 VeroScript::Audio SDL3::SDL3 Threads::Threads)
		target_compile_definitions(audio PRIVATE VS_TEST_AUDIO=1)
		add_test(NAME Audio COMMAND audio)
		set_tests_properties(Audio PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
		if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
			add_test(NAME AudioDisconnect COMMAND audio disconnect)
			set_tests_properties(AudioDisconnect PROPERTIES TIMEOUT 10 SKIP_RETURN_CODE 77
				ENVIRONMENT "SDL_AUDIODRIVER=disk;SDL_AUDIO_DISK_OUTPUT_FILE=/dev/full")
		endif()
	endif()
	if(VeroScriptSdl3_SDL3_GRAPHICS)
		foreach(_vs_case Graphics2D Composition2D GraphicsAsync)
			add_executable(${_vs_case} "${VEROSCRIPT_FOUNDATION_SOURCE_SDL3}/Resources/Tests/${_vs_case}Tests.cpp")
			target_link_libraries(${_vs_case} PRIVATE VeroScript::Sdl3)
			add_test(NAME ${_vs_case} COMMAND ${_vs_case})
			set_tests_properties(${_vs_case} PROPERTIES TIMEOUT 60)
		endforeach()
		if(VEROSCRIPT_TEST_BINDINGS AND TARGET VeroScript::GraphicsBindings)
			add_executable(graphics-script "${VEROSCRIPT_FOUNDATION_SOURCE_SDL3}/Resources/Tests/ScriptTests.cpp")
			target_link_libraries(graphics-script PRIVATE VeroScript::GraphicsBindings VeroScript::Sdl3)
			add_test(NAME ScriptGraphics COMMAND graphics-script)
			set_tests_properties(ScriptGraphics PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy")
		endif()
	endif()
	if(VEROSCRIPT_TEST_BINDINGS AND VeroScriptSdl3_NATIVE_EXTENSION AND TARGET VeroScript::Graphics)
		add_executable(provider "${VEROSCRIPT_FOUNDATION_SOURCE_SDL3}/Resources/Tests/ProviderTests.cpp")
		target_link_libraries(provider PRIVATE VeroScript::Core VeroScript::Platform VeroScript::Graphics)
		target_compile_definitions(provider PRIVATE VS_TEST_AUDIO=$<BOOL:${VeroScriptSdl3_SDL3_AUDIO}>)
		if(VeroScriptSdl3_SDL3_AUDIO)
			target_link_libraries(provider PRIVATE VeroScript::Audio)
		endif()
		if(MINGW)
			target_link_options(provider PRIVATE -municode)
		endif()
		add_test(NAME NativeProvider COMMAND provider "$<TARGET_FILE:VeroScript::Sdl3>")
		set_tests_properties(NativeProvider PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
	endif()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../CopyRuntime.cmake")
get_property(_vs_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(_vs_target IN LISTS _vs_targets)
	if(MSVC)
		target_compile_options(${_vs_target} PRIVATE /EHsc /utf-8)
	endif()
	if(WIN32 AND TARGET VeroScript::Sdl3)
		set(_vs_runtime_files)
		foreach(_vs_dependency VeroScript::Sdl3 SDL3::SDL3)
			list(APPEND _vs_runtime_files "$<TARGET_FILE:${_vs_dependency}>")
		endforeach()
		if(VeroScriptSdl3_SDL3_GRAPHICS)
			get_target_property(_vs_location VeroScript::Sdl3 IMPORTED_LOCATION_RELEASE)
			get_filename_component(_vs_bin "${_vs_location}" DIRECTORY)
			file(GLOB _vs_dlls "${_vs_bin}/*.dll")
			list(APPEND _vs_runtime_files ${_vs_dlls})
		endif()
		list(REMOVE_DUPLICATES _vs_runtime_files)
		veroscript_consumer_copy_runtime(${_vs_target} ${_vs_runtime_files})
	endif()
endforeach()
