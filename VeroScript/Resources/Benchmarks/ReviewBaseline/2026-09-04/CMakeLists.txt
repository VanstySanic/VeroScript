cmake_minimum_required(VERSION 3.25)
project(PerformanceReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(review review.cpp)
target_include_directories(review PRIVATE "${REPO}/Script/VeroScript/Public/Script" "${REPO}/Script/VeroScript/Public")
target_compile_definitions(review PRIVATE VEROSCRIPT_HAS_ASMJIT=1 VEROSCRIPT_ASMJIT_X64=1 VEROSCRIPT_ASMJIT_A64=0 VEROSCRIPT_ASMJIT_X86=0)
target_link_libraries(review PRIVATE "${REPO}/build/windows-msvc-vs2026/lib/Release/VeroScript.lib" "${REPO}/build/windows-msvc-vs2026/lib/Release/asmjit.lib")
target_compile_options(review PRIVATE /O2 /utf-8 /bigobj)
target_link_options(review PRIVATE /STACK:16777216)
add_executable(compare compare.cpp)
target_compile_options(compare PRIVATE /O2 /fp:precise /utf-8)
add_executable(cross cross.cpp)
target_include_directories(cross PRIVATE "${REPO}/Script/VeroScript/Public/Script" "${REPO}/Script/VeroScript/Public")
target_compile_definitions(cross PRIVATE VEROSCRIPT_HAS_ASMJIT=1 VEROSCRIPT_ASMJIT_X64=1 VEROSCRIPT_ASMJIT_A64=0 VEROSCRIPT_ASMJIT_X86=0)
target_link_libraries(cross PRIVATE "${REPO}/build/windows-msvc-vs2026/lib/Release/VeroScript.lib" "${REPO}/build/windows-msvc-vs2026/lib/Release/asmjit.lib")
target_compile_options(cross PRIVATE /O2 /utf-8 /bigobj)
target_link_options(cross PRIVATE /STACK:16777216)
foreach(kind escaping matrix)
  add_executable(aot_${kind} aot-wrapper.cpp)
  target_compile_definitions(aot_${kind} PRIVATE AOT_FILE="${kind}-aot.cpp" AOT_FUNCTION=vs_work_mcross_2F${kind}_2Evs)
  target_compile_options(aot_${kind} PRIVATE /O2 /fp:precise /utf-8 /bigobj)
endforeach()
add_executable(host_overhead host-overhead.cpp)
target_include_directories(host_overhead PRIVATE "${REPO}/Script/VeroScript/Public/Script" "${REPO}/Script/VeroScript/Public")
target_compile_definitions(host_overhead PRIVATE VEROSCRIPT_HAS_ASMJIT=1 VEROSCRIPT_ASMJIT_X64=1 VEROSCRIPT_ASMJIT_A64=0 VEROSCRIPT_ASMJIT_X86=0)
target_link_libraries(host_overhead PRIVATE "${REPO}/build/windows-msvc-vs2026/lib/Release/VeroScript.lib" "${REPO}/build/windows-msvc-vs2026/lib/Release/asmjit.lib")
target_compile_options(host_overhead PRIVATE /O2 /utf-8)
target_link_options(host_overhead PRIVATE /STACK:16777216)
